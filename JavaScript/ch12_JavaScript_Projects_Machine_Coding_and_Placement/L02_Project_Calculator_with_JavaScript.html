Lesson 02: Project – Calculator with JavaScript

Develop an interactive calculator using JavaScript functions, event handling, arithmetic operators, input validation and expression evaluation.

1. Learning Objectives

  • Build a responsive calculator interface using HTML and CSS.
  • Handle button clicks and keyboard input.
  • Perform arithmetic operations using JavaScript.
  • Understand expression evaluation and operator precedence.
  • Handle division by zero and invalid expressions.
  • Implement clear, delete and decimal operations.
  • Format numerical results and handle floating-point precision.
  • Practise calculator-related frontend machine-coding questions.

3. JavaScript Concepts Used

3.1 Arithmetic Operators

JavaScript provides arithmetic operators for calculations.

let a = 20;
let b = 5;

console.log(a + b); // 25
console.log(a - b); // 15
console.log(a * b); // 100
console.log(a / b); // 4
console.log(a % b); // 0

3.2 Event Listeners

Event listeners allow calculator buttons to respond to user actions. 

const button = document.querySelector("#calculate");

button.addEventListener("click", function () {
    console.log("Calculate button clicked");
});

3.3 Functions

Functions separate calculator operations into reusable units. 

function add(a, b) {
    return a + b;
}

function subtract(a, b) {
    return a - b;
}

console.log(add(10, 5));      // 15
console.log(subtract(10, 5)); // 5

3.4 Number Conversion

User input is initially received as text. Convert it into a number before performing arithmetic. 

const input = "25";
const value = Number(input);

console.log(value + 5); // 30

3.5 Floating-Point Precision

JavaScript uses IEEE 754 binary floating-point numbers. Some decimal calculations therefore produce small precision errors. 

console.log(0.1 + 0.2);
// 0.30000000000000004

console.log(Number((0.1 + 0.2).toFixed(10)));
// 0.3

3.6 Operator Precedence

Multiplication and division have higher precedence than addition and subtraction. Parentheses can change evaluation order. 

console.log(10 + 5 * 2);   // 20
console.log((10 + 5) * 2); // 30

A calculator can be implemented using either an operator-state approach or an expression-based approach. This project uses a controlled expression parser so that it can support parentheses and normal arithmetic precedence without using JavaScript eval().

<script>
    const display = document.getElementById("display");
    const message = document.getElementById("message");
    const buttons = document.querySelector(".buttons");

    let expression = "";
    let resultShown = false;

    // Display the current expression
    function updateDisplay() {
        display.value = expression || "0";
    }

    // Clear the calculator
    function clearCalculator() {
        expression = "";
        resultShown = false;
        message.textContent = "";
        updateDisplay();
    }

    // Delete the last character
    function deleteLast() {
        if (resultShown) {
            clearCalculator();
            return;
        }

        expression = expression.slice(0, -1);
        message.textContent = "";
        updateDisplay();
    }

    // Check whether a character is an operator
    function isOperator(char) {
        return ["+", "-", "*", "/", "%"].includes(char);
    }

    // Append a character to the expression
    function appendValue(value) {
        if (resultShown) {
            if (isOperator(value)) {
                resultShown = false;
            } else {
                expression = "";
                resultShown = false;
            }
        }

        const last = expression.slice(-1);

        if (value === ".") {
            const currentNumber =
                expression.split(/[+*/%()-]/).pop();

            if (currentNumber.includes(".")) {
                return;
            }

            if (
                currentNumber === "" ||
                currentNumber === undefined
            ) {
                expression += "0";
            }
        }

        if (isOperator(value)) {
            if (expression === "" && value !== "-") {
                return;
            }

            if (isOperator(last)) {
                if (value === "-" && last !== "-") {
                    expression += value;
                } else {
                    expression =
                        expression.slice(0, -1) + value;
                }

                updateDisplay();
                return;
            }
        }

        if (value === ")") {
            const openCount =
                (expression.match(/\(/g) || []).length;

            const closeCount =
                (expression.match(/\)/g) || []).length;

            if (
                openCount <= closeCount ||
                !/[0-9)]$/.test(expression)
            ) {
                return;
            }
        }

        if (
            value === "(" &&
            /[0-9)]$/.test(expression)
        ) {
            expression += "*";
        }

        if (
            /[0-9]/.test(value) &&
            last === ")"
        ) {
            expression += "*";
        }

        expression += value;

        message.textContent = "";
        updateDisplay();
    }

    // Tokenize a mathematical expression
    function tokenize(input) {
        const tokens = [];
        let index = 0;

        while (index < input.length) {
            const char = input[index];

            if (/\s/.test(char)) {
                index++;
                continue;
            }

            if (/[0-9.]/.test(char)) {
                const start = index;
                let decimalCount = 0;

                while (
                    index < input.length &&
                    /[0-9.]/.test(input[index])
                ) {
                    if (input[index] === ".") {
                        decimalCount++;
                    }

                    index++;
                }

                const numberText = input.slice(start, index);

                if (
                    decimalCount > 1 ||
                    numberText === "." ||
                    !Number.isFinite(Number(numberText))
                ) {
                    throw new Error("Invalid number");
                }

                tokens.push(Number(numberText));
                continue;
            }

            if ("+-*/%()".includes(char)) {
                tokens.push(char);
                index++;
                continue;
            }

            throw new Error("Invalid character");
        }

        return tokens;
    }

    // Recursive-descent expression parser
    function evaluateExpression(input) {
        const tokens = tokenize(input);
        let position = 0;

        function current() {
            return tokens[position];
        }

        function parsePrimary() {
            const token = current();

            if (token === "+") {
                position++;
                return parsePrimary();
            }

            if (token === "-") {
                position++;
                return -parsePrimary();
            }

            if (token === "(") {
                position++;

                const value = parseAddition();

                if (current() !== ")") {
                    throw new Error("Missing closing parenthesis");
                }

                position++;
                return value;
            }

            if (typeof token === "number") {
                position++;
                return token;
            }

            throw new Error("Expected a number");
        }

        function parseMultiplication() {
            let value = parsePrimary();

            while (
                current() === "*" ||
                current() === "/" ||
                current() === "%"
            ) {
                const operator = current();
                position++;

                const right = parsePrimary();

                if (
                    (operator === "/" || operator === "%") &&
                    right === 0
                ) {
                    throw new Error("Cannot divide by zero");
                }

                if (operator === "*") {
                    value *= right;
                } else if (operator === "/") {
                    value /= right;
                } else {
                    value %= right;
                }
            }

            return value;
        }

        function parseAddition() {
            let value = parseMultiplication();

            while (
                current() === "+" ||
                current() === "-"
            ) {
                const operator = current();
                position++;

                const right = parseMultiplication();

                if (operator === "+") {
                    value += right;
                } else {
                    value -= right;
                }
            }

            return value;
        }

        if (tokens.length === 0) {
            throw new Error("Enter an expression");
        }

        const result = parseAddition();

        if (position !== tokens.length) {
            throw new Error("Invalid expression");
        }

        if (!Number.isFinite(result)) {
            throw new Error("Result is outside the supported range");
        }

        return Number(result.toPrecision(12));
    }

    // Calculate the result
    function calculate() {
        if (!expression) {
            return;
        }

        try {
            const result = evaluateExpression(expression);

            expression = String(result);
            resultShown = true;

            message.textContent = "Calculation successful.";
            updateDisplay();

        } catch (error) {
            message.textContent = error.message;
        }
    }

    // Handle calculator button clicks
    buttons.addEventListener("click", event => {
        const button = event.target.closest("button");

        if (!button) {
            return;
        }

        const action = button.dataset.action;
        const value = button.dataset.value;

        if (action === "clear") {
            clearCalculator();
        } else if (action === "delete") {
            deleteLast();
        } else if (action === "calculate") {
            calculate();
        } else if (value !== undefined) {
            appendValue(value);
        }
    });

    // Handle keyboard input
    document.addEventListener("keydown", event => {
        const key = event.key;

        if (/^[0-9.+\-*/%()]$/.test(key)) {
            event.preventDefault();
            appendValue(key);
        } else if (key === "Enter" || key === "=") {
            event.preventDefault();
            calculate();
        } else if (key === "Backspace") {
            event.preventDefault();
            deleteLast();
        } else if (key === "Escape") {
            clearCalculator();
        }
    });

    updateDisplay();
</script>

</body>
</html>

7. Understanding the Expression Parser

The project uses a recursive-descent parser. This is a structured way to evaluate mathematical expressions without using eval() or Function().

Parser Function Responsibility
tokenize() Converts expression text into numbers and operators.
parsePrimary() Handles numbers, unary signs and parentheses.
parseMultiplication() Handles multiplication, division and remainder.
parseAddition() Handles addition and subtraction.
evaluateExpression() Validates the complete expression and returns the result.

Why Is This Better Than eval()?

  • It accepts only the supported mathematical tokens.
  • It rejects unexpected characters.
  • It explicitly handles operator precedence.
  • It provides controlled error handling.
  • It avoids executing arbitrary JavaScript code.

Step 2: 10 + 10 = 20

10. Common Mistakes and Debugging

  • Using eval() directly: Avoid evaluating untrusted input as JavaScript code.
  • Ignoring operator precedence: Evaluate multiplication and division before addition and subtraction.
  • Not converting input: Convert text into numbers before arithmetic operations.
  • Division by zero: Validate the divisor before calculating.
  • Floating-point precision: Format results when displaying decimal calculations.
  • Incorrect event targets: Use closest("button") when handling clicks on nested button content.
  • Missing error handling: Catch invalid expressions and show useful messages.

11. Placement Interview Questions and Answers

Q1. What are arithmetic operators in JavaScript?

Arithmetic operators perform mathematical calculations. Examples include +, -, *, /, % and **.

Q2. What is operator precedence?

Operator precedence determines the order in which operators are evaluated in an expression.

Q3. What is the difference between parseInt() and Number()?

parseInt() parses an integer from the beginning of a string, while Number() attempts to convert the entire value into a number.

Q4. What is NaN?

NaN means Not-a-Number. It represents an invalid numerical result. Use Number.isNaN() to check for it.

Q5. Why does 0.1 + 0.2 sometimes produce an unexpected result?

Binary floating-point representation cannot exactly represent every decimal fraction.

Q6. What is event delegation?

Event delegation attaches one event listener to a parent element to handle events from its child elements using event bubbling.

Q7. What does preventDefault() do?

It prevents the browser's default action for an event when that action should be replaced with custom behaviour.

Q8. What is the purpose of try/catch?

It handles exceptions so that an application can respond to errors instead of terminating the current operation unexpectedly.

Q9. What is a recursive-descent parser?

It is a parsing technique that uses functions to process different grammatical rules of an expression.

Q10. Why should eval() be avoided?

eval() executes a string as JavaScript code and can create serious security and maintainability problems when input is not trusted.

Q11. What is the difference between == and ===?

== allows type coercion during comparison. === compares values without type coercion.

Q12. What is the remainder operator?

The % operator returns the remainder after division. For example, 10 % 3 returns 1.

Q13. What is the purpose of Number.isFinite()?

It checks whether a value is a finite number, excluding Infinity, -Infinity and NaN.

Q14. What is a unary operator?

A unary operator operates on one operand. Examples include unary plus (+value) and unary minus (-value).

Q15. What is the difference between keydown and keyup?

keydown fires when a key is pressed. keyup fires when the key is released.
