Lesson 10: Implementing a Basic Character Driver
Level: Intermediate – Practical
Prerequisites: Character device registration, cdev_init(), cdev_add(), file_operations, open() and release().
Objective: Develop, compile, load, test and unload a basic Linux character-device driver.

1. Introduction

In the previous lessons, we studied the individual components required to implement a character-device driver.

These components include:

  • Device-number allocation.
  • struct cdev.
  • cdev_init().
  • cdev_add().
  • struct file_operations.
  • Driver open callback.
  • Driver release callback.
  • Device-number cleanup.

In this practical lesson, we combine these concepts into a working character driver that supports reading and writing data.

2. What Are We Going to Build?

We will implement a simple memory-backed character device. It will store data in a kernel buffer and allow user-space applications to read and write it.

Driver Features

Feature Description
Device registration Allocate one major/minor device-number range and register a cdev.
Open operation Handle applications opening the device.
Read operation Copy stored data from kernel space to user space.
Write operation Copy data from user space into the kernel buffer.
Release operation Handle final release of an open-file instance.
Cleanup Remove cdev registration and release allocated resources.
This is a software-based demonstration driver. It does not control physical hardware. It is suitable for learning character-driver fundamentals in an Ubuntu virtual machine.

6. Complete Character Driver Source Code

Create the file basic_char_driver.c. 

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/slab.h>
#include <linux/uaccess.h>
#include <linux/mutex.h>
#include <linux/string.h>
#include <linux/errno.h>

#define DEVICE_NAME "mychardev"
#define BUFFER_SIZE 1024

static dev_t dev;
static struct cdev my_cdev;

static char *device_buffer;
static size_t data_size;

static DEFINE_MUTEX(device_mutex);

/* Open callback */
static int my_open(struct inode *inode,
                   struct file *file)
{
    pr_info("mychardev: device opened\n");

    return 0;
}

/* Read callback */
static ssize_t my_read(
    struct file *file,
    char __user *user_buffer,
    size_t count,
    loff_t *position
)
{
    size_t bytes_to_copy;
    size_t remaining;
    ssize_t ret;

    if (mutex_lock_interruptible(&device_mutex))
        return -ERESTARTSYS;

    if (*position < 0) {
        ret = -EINVAL;
        goto out;
    }

    if (*position >= data_size) {
        ret = 0;
        goto out;
    }

    remaining = data_size - (size_t)*position;

    bytes_to_copy = min(count, remaining);

    if (copy_to_user(
            user_buffer,
            device_buffer + (size_t)*position,
            bytes_to_copy)) {
        ret = -EFAULT;
        goto out;
    }

    *position += bytes_to_copy;

    ret = bytes_to_copy;

    pr_info("mychardev: read %zu bytes\n",
            bytes_to_copy);

out:
    mutex_unlock(&device_mutex);
    return ret;
}

/* Write callback */
static ssize_t my_write(
    struct file *file,
    const char __user *user_buffer,
    size_t count,
    loff_t *position
)
{
    size_t bytes_to_copy;
    size_t available;
    ssize_t ret;

    if (mutex_lock_interruptible(&device_mutex))
        return -ERESTARTSYS;

    if (*position < 0 ||
        *position >= BUFFER_SIZE) {
        ret = -ENOSPC;
        goto out;
    }

    available = BUFFER_SIZE - (size_t)*position;

    bytes_to_copy = min(count, available);

    if (bytes_to_copy == 0) {
        ret = -ENOSPC;
        goto out;
    }

    if (copy_from_user(
            device_buffer + (size_t)*position,
            user_buffer,
            bytes_to_copy)) {
        ret = -EFAULT;
        goto out;
    }

    *position += bytes_to_copy;

    if ((size_t)*position > data_size)
        data_size = (size_t)*position;

    ret = bytes_to_copy;

    pr_info("mychardev: wrote %zu bytes\n",
            bytes_to_copy);

out:
    mutex_unlock(&device_mutex);
    return ret;
}

/* Release callback */
static int my_release(struct inode *inode,
                      struct file *file)
{
    pr_info("mychardev: device released\n");

    return 0;
}

/* File operations */
static const struct file_operations my_fops = {
    .owner = THIS_MODULE,
    .open = my_open,
    .read = my_read,
    .write = my_write,
    .release = my_release,
};

/* Module initialization */
static int __init my_driver_init(void)
{
    int ret;

    device_buffer = kzalloc(BUFFER_SIZE, GFP_KERNEL);

    if (!device_buffer) {
        pr_err("mychardev: buffer allocation failed\n");
        return -ENOMEM;
    }

    data_size = 0;

    ret = alloc_chrdev_region(
        &dev,
        0,
        1,
        DEVICE_NAME
    );

    if (ret < 0) {
        pr_err("mychardev: device number allocation failed\n");
        kfree(device_buffer);
        return ret;
    }

    cdev_init(&my_cdev, &my_fops);
    my_cdev.owner = THIS_MODULE;

    ret = cdev_add(&my_cdev, dev, 1);

    if (ret < 0) {
        pr_err("mychardev: cdev_add failed: %d\n", ret);

        unregister_chrdev_region(dev, 1);
        kfree(device_buffer);

        return ret;
    }

    pr_info("mychardev: driver loaded\n");
    pr_info("mychardev: major=%u minor=%u\n",
            MAJOR(dev), MINOR(dev));

    return 0;
}

/* Module cleanup */
static void __exit my_driver_exit(void)
{
    cdev_del(&my_cdev);

    unregister_chrdev_region(dev, 1);

    kfree(device_buffer);

    pr_info("mychardev: driver unloaded\n");
}

module_init(my_driver_init);
module_exit(my_driver_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("NS Learning Hub");
MODULE_DESCRIPTION("Basic Memory-Backed Character Device Driver");

7. Understanding the Source Code

A. Header Files

Header Purpose
linux/module.h Kernel module support.
linux/kernel.h Common kernel definitions and helpers.
linux/init.h Module initialization and exit annotations.
linux/fs.h File operations and file-system interfaces.
linux/cdev.h Character-device registration.
linux/slab.h Kernel memory allocation.
linux/uaccess.h User-space data transfer functions.
linux/mutex.h Mutex synchronization.

B. Global Variables

static dev_t dev;
static struct cdev my_cdev;

static char *device_buffer;
static size_t data_size;

static DEFINE_MUTEX(device_mutex);
  • dev stores the allocated device number.
  • my_cdev represents the character device.
  • device_buffer points to the kernel data buffer.
  • data_size stores the number of valid data bytes.
  • device_mutex protects shared buffer operations.

C. Open Callback

static int my_open(struct inode *inode,
                   struct file *file)
{
    pr_info("mychardev: device opened\n");
    return 0;
}

This callback reports that the device has been opened. It does not allocate additional per-open resources.

D. Read Callback

if (copy_to_user(
        user_buffer,
        device_buffer + (size_t)*position,
        bytes_to_copy)) {
    ret = -EFAULT;
    goto out;
}

copy_to_user() transfers data from the kernel buffer to the application buffer.

The read callback also updates the file position and returns the number of bytes transferred.

E. Write Callback

if (copy_from_user(
        device_buffer + (size_t)*position,
        user_buffer,
        bytes_to_copy)) {
    ret = -EFAULT;
    goto out;
}

copy_from_user() transfers data from user space into the kernel buffer.

The callback checks the buffer boundary and updates the valid data size after a successful transfer.

F. Release Callback

static int my_release(struct inode *inode,
                      struct file *file)
{
    pr_info("mychardev: device released\n");
    return 0;
}

This callback reports final release of the open-file instance. The shared device buffer remains allocated until module cleanup.

G. File Operations

static const struct file_operations my_fops = {
    .owner = THIS_MODULE,
    .open = my_open,
    .read = my_read,
    .write = my_write,
    .release = my_release,
};

This structure connects the kernel's file-operation requests to the driver callbacks.

8. Create the Makefile

Create a file named Makefile. 

obj-m += basic_char_driver.o

all:
	make -C /lib/modules/$(shell uname -r)/build M=$(PWD) modules

clean:
	make -C /lib/modules/$(shell uname -r)/build M=$(PWD) clean
The recipe commands under all: and clean: must start with a tab character, not spaces.

9. Install Build Dependencies

Before compiling the driver, install the required packages. 

sudo apt update

sudo apt install build-essential \
linux-headers-$(uname -r)

Verify that the kernel build directory exists: 

ls /lib/modules/$(uname -r)/build

10. Compile the Character Driver

From the project directory, execute: 

make

If compilation is successful, the kernel module file basic_char_driver.ko will be generated.

Inspect the generated module:

ls -l

Check module information:

modinfo basic_char_driver.ko
Always compile against the kernel headers corresponding to the kernel you intend to run the module on.

11. Load the Kernel Module

Load the module using: 

sudo insmod basic_char_driver.ko

Check whether it is loaded: 

lsmod | grep basic_char_driver

Inspect the kernel log: 

sudo dmesg | tail -20

Example output:

mychardev: driver loaded
mychardev: major=240 minor=0

The major number is assigned dynamically. Your system may show a different value.

12. Create the Device Node

The driver has registered a character device, but it has not created a device node in /dev.

We can create a temporary node manually using mknod.

Step 1: Obtain the Major Number

sudo dmesg | tail -20

Note the major and minor numbers printed by the driver.

Step 2: Create the Device Node

Suppose the allocated major number is 240 and the minor number is 0. Use: 

sudo mknod /dev/mychardev c 240 0

Replace 240 with the actual major number reported by your driver.

Step 3: Set Device Permissions

sudo chmod 666 /dev/mychardev

Step 4: Verify the Device Node

ls -l /dev/mychardev

The first character in the permission display should be c, indicating a character device.

The manual node is only a temporary testing interface. In a production driver, device-model registration using class_create() and device_create() is commonly used so that devtmpfs/udev can provide the device node.

13. Test the Write Operation

Write a string into the device: 

echo "Hello Linux Device Driver" | sudo tee /dev/mychardev

The write callback receives the user-space data and copies it into the kernel buffer.

Inspect kernel messages: 

sudo dmesg | tail -20

Expected message includes:

mychardev: device opened
mychardev: wrote 26 bytes
mychardev: device released

The exact byte count depends on the text and newline written.

14. Test the Read Operation

Read the data from the device: 

sudo cat /dev/mychardev

Expected output:

Hello Linux Device Driver

The read callback transfers the stored kernel-buffer contents into the user-space read buffer.

The demonstration uses a shared kernel buffer. A new open file normally starts with its own file position at zero. This allows another application to read the stored data from the beginning.

15. Test Using a C User-Space Program

We can also access the device through a normal C application.

File: test_driver.c

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>

int main(void)
{
    int fd;
    char write_data[] = "Hello from user space";
    char read_buffer[128] = {0};
    ssize_t bytes;

    fd = open("/dev/mychardev", O_RDWR);

    if (fd < 0) {
        perror("open failed");
        return 1;
    }

    bytes = write(fd, write_data,
                  strlen(write_data));

    if (bytes < 0) {
        perror("write failed");
        close(fd);
        return 1;
    }

    printf("Written bytes: %zd\n", bytes);

    if (lseek(fd, 0, SEEK_SET) < 0) {
        perror("lseek failed");
        close(fd);
        return 1;
    }

    bytes = read(fd, read_buffer,
                 sizeof(read_buffer) - 1);

    if (bytes < 0) {
        perror("read failed");
        close(fd);
        return 1;
    }

    read_buffer[bytes] = '\0';

    printf("Read data: %s\n", read_buffer);

    close(fd);

    return 0;
}

Compile the User-Space Program

gcc test_driver.c -o test_driver

Run the Program

sudo ./test_driver

Expected output:

Written bytes: 21
Read data: Hello from user space

The program performs the following:

  1. Opens the character device.
  2. Writes a string to the device.
  3. Moves the file position to the beginning using lseek().
  4. Reads the stored string.
  5. Closes the device file.
This driver does not implement a custom .llseek callback. The default seek behavior is sufficient for this simple memory-backed example. The read and write callbacks validate positions and buffer boundaries.

17. Remove the Device Node and Unload the Driver

Before unloading the module, close all applications using the device.

Step 1: Remove the Device Node

sudo rm /dev/mychardev

Step 2: Unload the Module

sudo rmmod basic_char_driver

Step 3: Verify Unloading

lsmod | grep basic_char_driver

Step 4: Check Kernel Messages

sudo dmesg | tail -20

Step 5: Clean the Project

make clean
Do not unload the module while applications are still using the device. The .owner = THIS_MODULE reference normally prevents ordinary unloading while open file operations are active. The driver must also manage shared-resource lifetime correctly.

18. Common Errors and Troubleshooting

Error Possible Cause and Solution
Invalid module format Check the running kernel version and rebuild against its matching headers.
Operation not permitted Use appropriate administrator privileges for module operations.
Device or resource busy Check whether the device is already in use or the module is still referenced.
No such file or directory Verify that /dev/mychardev exists.
Permission denied Check device-node permissions and access privileges.
Wrong major number Use the major number printed by the currently loaded driver.
Read returns no data Check whether data was written and whether the file position is at the end.
Compilation error Check kernel headers, function signatures and source-code syntax.
