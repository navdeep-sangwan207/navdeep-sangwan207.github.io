1. Project Introduction

In this lesson, we will develop a complete Linux character device driver and a user-space application to communicate with it.

The driver will provide the following functionality:

  • Register a character device dynamically.
  • Create a device class and device node.
  • Implement open(), read(), write() and release().
  • Implement IOCTL commands.
  • Use kernel memory allocation.
  • Protect shared data using a mutex.
  • Transfer data safely between kernel space and user space.
  • Build a C application to test all driver operations.

4. Define IOCTL Commands

IOCTL allows a user-space application to perform device-specific control operations.

Create the shared header file:

nano my_driver_ioctl.h

File: my_driver_ioctl.h

#ifndef MY_DRIVER_IOCTL_H
#define MY_DRIVER_IOCTL_H

#include <linux/ioctl.h>

#define MY_MAGIC 'M'

#define MY_IOCTL_CLEAR \
    _IO(MY_MAGIC, 1)

#define MY_IOCTL_GET_SIZE \
    _IOR(MY_MAGIC, 2, int)

#define MY_IOCTL_SET_VALUE \
    _IOW(MY_MAGIC, 3, int)

#define MY_IOCTL_GET_VALUE \
    _IOR(MY_MAGIC, 4, int)

#endif

Command meanings:

CommandPurpose
MY_IOCTL_CLEARClear the device buffer.
MY_IOCTL_GET_SIZEGet the current buffer length.
MY_IOCTL_SET_VALUESet an integer value.
MY_IOCTL_GET_VALUERead the stored integer value.
The header is shared between kernel and user space. In a production project, use fixed-width UAPI types such as __u32 and __s32 for data structures and carefully design ABI-compatible IOCTL interfaces.

5. Complete Character Device Driver

Create the kernel source file:

nano complete_driver.c

File: complete_driver.c

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/slab.h>
#include <linux/uaccess.h>
#include <linux/mutex.h>
#include <linux/string.h>

#include "my_driver_ioctl.h"

#define DEVICE_NAME "my_device"
#define CLASS_NAME  "my_device_class"
#define BUFFER_SIZE 1024

static dev_t dev_number;
static struct cdev my_cdev;
static struct class *my_class;
static struct device *my_device;

static char *device_buffer;
static size_t buffer_length;

static int stored_value;

static DEFINE_MUTEX(device_mutex);

/* Open callback */
static int my_open(struct inode *inode,
                   struct file *file)
{
    pr_info("my_device: opened\n");
    return 0;
}

/* Read callback */
static ssize_t my_read(struct file *file,
                       char __user *user_buffer,
                       size_t count,
                       loff_t *offset)
{
    size_t available;
    size_t bytes_to_copy;
    ssize_t ret;

    if (mutex_lock_interruptible(&device_mutex))
        return -ERESTARTSYS;

    if (*offset >= buffer_length) {
        ret = 0;
        goto out;
    }

    available = buffer_length - *offset;

    bytes_to_copy = min(count, available);

    if (copy_to_user(
            user_buffer,
            device_buffer + *offset,
            bytes_to_copy)) {
        ret = -EFAULT;
        goto out;
    }

    *offset += bytes_to_copy;
    ret = bytes_to_copy;

out:
    mutex_unlock(&device_mutex);
    return ret;
}

/* Write callback */
static ssize_t my_write(struct file *file,
                        const char __user *user_buffer,
                        size_t count,
                        loff_t *offset)
{
    size_t bytes_to_copy;
    ssize_t ret;

    if (mutex_lock_interruptible(&device_mutex))
        return -ERESTARTSYS;

    bytes_to_copy = min(count, (size_t)BUFFER_SIZE);

    if (copy_from_user(
            device_buffer,
            user_buffer,
            bytes_to_copy)) {
        ret = -EFAULT;
        goto out;
    }

    buffer_length = bytes_to_copy;

    *offset = bytes_to_copy;

    ret = bytes_to_copy;

    pr_info("my_device: received %zu bytes\n",
            bytes_to_copy);

out:
    mutex_unlock(&device_mutex);
    return ret;
}

/* IOCTL callback */
static long my_ioctl(struct file *file,
                     unsigned int cmd,
                     unsigned long arg)
{
    int value;
    int ret = 0;

    switch (cmd) {

    case MY_IOCTL_CLEAR:

        if (mutex_lock_interruptible(&device_mutex))
            return -ERESTARTSYS;

        memset(device_buffer, 0, BUFFER_SIZE);
        buffer_length = 0;

        mutex_unlock(&device_mutex);

        pr_info("my_device: buffer cleared\n");
        break;

    case MY_IOCTL_GET_SIZE:

        if (mutex_lock_interruptible(&device_mutex))
            return -ERESTARTSYS;

        value = (int)buffer_length;

        mutex_unlock(&device_mutex);

        if (copy_to_user(
                (int __user *)arg,
                &value,
                sizeof(value)))
            ret = -EFAULT;

        break;

    case MY_IOCTL_SET_VALUE:

        if (copy_from_user(
                &value,
                (int __user *)arg,
                sizeof(value)))
            return -EFAULT;

        if (mutex_lock_interruptible(&device_mutex))
            return -ERESTARTSYS;

        stored_value = value;

        mutex_unlock(&device_mutex);

        pr_info("my_device: value set to %d\n",
                value);
        break;

    case MY_IOCTL_GET_VALUE:

        if (mutex_lock_interruptible(&device_mutex))
            return -ERESTARTSYS;

        value = stored_value;

        mutex_unlock(&device_mutex);

        if (copy_to_user(
                (int __user *)arg,
                &value,
                sizeof(value)))
            ret = -EFAULT;

        break;

    default:
        ret = -ENOTTY;
        break;
    }

    return ret;
}

/* Release callback */
static int my_release(struct inode *inode,
                      struct file *file)
{
    pr_info("my_device: closed\n");
    return 0;
}

/* File operations */
static const struct file_operations my_fops = {
    .owner = THIS_MODULE,
    .open = my_open,
    .read = my_read,
    .write = my_write,
    .unlocked_ioctl = my_ioctl,
    .release = my_release,
};

/* Module initialization */
static int __init my_driver_init(void)
{
    int ret;

    device_buffer = kzalloc(
        BUFFER_SIZE,
        GFP_KERNEL
    );

    if (!device_buffer)
        return -ENOMEM;

    ret = alloc_chrdev_region(
        &dev_number,
        0,
        1,
        DEVICE_NAME
    );

    if (ret)
        goto free_buffer;

    cdev_init(&my_cdev, &my_fops);
    my_cdev.owner = THIS_MODULE;

    ret = cdev_add(
        &my_cdev,
        dev_number,
        1
    );

    if (ret)
        goto unregister_region;

    my_class = class_create(CLASS_NAME);

    if (IS_ERR(my_class)) {
        ret = PTR_ERR(my_class);
        goto delete_cdev;
    }

    my_device = device_create(
        my_class,
        NULL,
        dev_number,
        NULL,
        DEVICE_NAME
    );

    if (IS_ERR(my_device)) {
        ret = PTR_ERR(my_device);
        goto destroy_class;
    }

    pr_info("my_device: driver initialized\n");
    pr_info("Major: %d Minor: %d\n",
            MAJOR(dev_number),
            MINOR(dev_number));

    return 0;

destroy_class:
    class_destroy(my_class);

delete_cdev:
    cdev_del(&my_cdev);

unregister_region:
    unregister_chrdev_region(dev_number, 1);

free_buffer:
    kfree(device_buffer);

    return ret;
}

/* Module cleanup */
static void __exit my_driver_exit(void)
{
    device_destroy(my_class, dev_number);
    class_destroy(my_class);

    cdev_del(&my_cdev);

    unregister_chrdev_region(
        dev_number,
        1
    );

    kfree(device_buffer);

    pr_info("my_device: driver removed\n");
}

module_init(my_driver_init);
module_exit(my_driver_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("NS Learning Hub");
MODULE_DESCRIPTION("Complete Linux Character Device Driver");
MODULE_VERSION("1.0");
Kernel compatibility: This example uses the modern class_create("name") API. Older kernels may use a different signature. Compile against the exact target kernel headers.

6. Explanation of Driver Functions

6.1 Dynamic Device Number Allocation

alloc_chrdev_region(
    &dev_number,
    0,
    1,
    DEVICE_NAME
);

Allocates a major and minor device number dynamically.

6.2 Character Device Registration

cdev_init(&my_cdev, &my_fops);
cdev_add(&my_cdev, dev_number, 1);

Initializes the character device and makes it available to the kernel's character-device subsystem.

6.3 Kernel Memory Allocation

device_buffer = kzalloc(
    BUFFER_SIZE,
    GFP_KERNEL
);

Allocates a zero-initialized kernel buffer.

6.4 User-to-Kernel Data Transfer

copy_from_user(
    device_buffer,
    user_buffer,
    bytes_to_copy
);

Copies data from a user-space buffer into kernel memory.

6.5 Kernel-to-User Data Transfer

copy_to_user(
    user_buffer,
    device_buffer,
    bytes_to_copy
);

Copies data from kernel memory to a user-space buffer.

6.6 Mutex Synchronization

mutex_lock_interruptible(&device_mutex);
mutex_unlock(&device_mutex);

Protects shared device state against concurrent access.

10. User-Space Test Application

Create a C application:

nano test_driver.c

File: test_driver.c

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>
#include <sys/ioctl.h>
#include <errno.h>

#include "my_driver_ioctl.h"

int main(void)
{
    int fd;
    int value;
    int size;
    char write_data[] = "Linux Device Driver Project";
    char read_data[128];
    ssize_t bytes;

    fd = open("/dev/my_device", O_RDWR);

    if (fd < 0) {
        perror("open");
        return 1;
    }

    printf("Device opened successfully\n");

    /* Write data */
    bytes = write(
        fd,
        write_data,
        strlen(write_data)
    );

    if (bytes < 0) {
        perror("write");
        goto error;
    }

    printf("Written: %s\n", write_data);

    /* Reset the file position before reading */
    if (lseek(fd, 0, SEEK_SET) < 0) {
        perror("lseek");
        goto error;
    }

    /* Read data */
    bytes = read(
        fd,
        read_data,
        sizeof(read_data) - 1
    );

    if (bytes < 0) {
        perror("read");
        goto error;
    }

    read_data[bytes] = '\0';

    printf("Read: %s\n", read_data);

    /* Get buffer size */
    if (ioctl(fd, MY_IOCTL_GET_SIZE, &size) < 0) {
        perror("GET_SIZE");
        goto error;
    }

    printf("Buffer size: %d\n", size);

    /* Set an integer value */
    value = 100;

    if (ioctl(fd, MY_IOCTL_SET_VALUE, &value) < 0) {
        perror("SET_VALUE");
        goto error;
    }

    printf("Stored value: %d\n", value);

    /* Read the stored integer value */
    value = 0;

    if (ioctl(fd, MY_IOCTL_GET_VALUE, &value) < 0) {
        perror("GET_VALUE");
        goto error;
    }

    printf("Retrieved value: %d\n", value);

    /* Clear the device buffer */
    if (ioctl(fd, MY_IOCTL_CLEAR) < 0) {
        perror("CLEAR");
        goto error;
    }

    printf("Device buffer cleared\n");

    close(fd);

    printf("Device closed\n");

    return 0;

error:
    close(fd);
    return 1;
}
The MY_IOCTL_CLEAR command uses _IO, so it does not require a third argument. The other IOCTL commands pass a pointer to an integer.

15. Driver Development Workflow

  1. Identify the hardware or device requirement.
  2. Select the appropriate Linux driver subsystem.
  3. Design the character-device or subsystem interface.
  4. Allocate and register device resources.
  5. Implement file operations and device callbacks.
  6. Handle memory allocation and synchronization.
  7. Compile against the target kernel headers.
  8. Load and test the kernel module.
  9. Validate error handling and concurrent access.
  10. Perform cleanup and reliability testing.

16. Important Design Considerations

  • Validate all user-provided inputs.
  • Never directly dereference user-space pointers.
  • Use suitable synchronization for shared state.
  • Check every allocation and registration return value.
  • Release resources in reverse order of acquisition.
  • Do not sleep while holding a spinlock or inside hard IRQ context.
  • Consider concurrent readers and writers.
  • Design IOCTL interfaces with long-term ABI compatibility in mind.
Production limitation: This is an educational memory-backed driver. It demonstrates the main character-device mechanisms but is not a production hardware driver. A production implementation should define the intended multi-process semantics, handle partial transfers and file-position rules carefully, consider 32-bit compatibility where needed, and test concurrent access and device removal.
