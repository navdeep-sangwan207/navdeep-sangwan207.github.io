1. Introduction to IOCTL

IOCTL stands for Input/Output Control. It is a mechanism used by user-space applications to send device-specific control commands to a kernel driver.

Normal read() and write() operations transfer data. However, many devices also require control operations such as:

  • Resetting a device.
  • Changing operating modes.
  • Setting configuration parameters.
  • Reading device status.
  • Starting or stopping an operation.

The Linux ioctl() system call provides an interface for these operations.

4. IOCTL Command Macros

Linux provides four commonly used macros in <linux/ioctl.h>.

MacroPurpose
_IOCommand without an associated data transfer.
_IORCommand for reading data from the driver into user space.
_IOWCommand for writing data from user space to the driver.
_IOWRCommand for bidirectional data transfer.

4.1 _IO

#define MYDEV_RESET _IO('M', 0)

Used for an operation such as resetting the device.

4.2 _IOR

#define MYDEV_GET_STATUS _IOR('M', 1, int)

Used when the driver returns an integer status to the application.

4.3 _IOW

#define MYDEV_SET_VALUE _IOW('M', 2, int)

Used when the application sends an integer configuration value to the driver.

4.4 _IOWR

#define MYDEV_EXCHANGE _IOWR('M', 3, int)

Used when the driver and application exchange data in both directions.

Important: The direction encoded in an IOCTL command describes data transfer from the application's perspective. The macros also encode command type, command number and data size. They do not automatically copy data; the driver must implement safe copying.

5. Shared IOCTL Header File

The kernel driver and user-space application must use the same command definitions. Create a shared header file.

nano my_ioctl.h

File: my_ioctl.h

#ifndef MY_IOCTL_H
#define MY_IOCTL_H

#ifdef __KERNEL__
#include <linux/ioctl.h>
#include <linux/types.h>
#else
#include <sys/ioctl.h>
#include <stdint.h>
#endif

#define MYDEV_MAGIC 'M'

#define MYDEV_RESET \
    _IO(MYDEV_MAGIC, 0)

#define MYDEV_GET_STATUS \
    _IOR(MYDEV_MAGIC, 1, int)

#define MYDEV_SET_VALUE \
    _IOW(MYDEV_MAGIC, 2, int)

#define MYDEV_EXCHANGE \
    _IOWR(MYDEV_MAGIC, 3, int)

#endif

This header provides consistent command numbers and definitions for both sides of the interface.

7. Complete Kernel Driver Source Code

Create the kernel module:

nano ioctl_driver.c

File: ioctl_driver.c

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/uaccess.h>
#include <linux/mutex.h>
#include <linux/errno.h>

#include "my_ioctl.h"

#define DEVICE_NAME "myioctl"

static dev_t dev_num;
static struct cdev my_cdev;

static int device_value = 0;
static int device_status = 1;

static DEFINE_MUTEX(device_mutex);

/* Open callback */
static int my_open(struct inode *inode, struct file *file)
{
    pr_info("myioctl: Device opened\n");
    return 0;
}

/* IOCTL callback */
static long my_ioctl(
    struct file *file,
    unsigned int cmd,
    unsigned long arg)
{
    int value;
    long ret = 0;

    if (_IOC_TYPE(cmd) != MYDEV_MAGIC)
        return -ENOTTY;

    if (_IOC_NR(cmd) > 3)
        return -ENOTTY;

    switch (cmd) {

    case MYDEV_RESET:

        if (mutex_lock_interruptible(&device_mutex))
            return -ERESTARTSYS;

        device_value = 0;
        device_status = 1;

        mutex_unlock(&device_mutex);

        pr_info("myioctl: Device reset\n");
        break;

    case MYDEV_GET_STATUS:

        if (mutex_lock_interruptible(&device_mutex))
            return -ERESTARTSYS;

        value = device_status;

        mutex_unlock(&device_mutex);

        if (copy_to_user((int __user *)arg,
                         &value,
                         sizeof(value)))
            ret = -EFAULT;

        break;

    case MYDEV_SET_VALUE:

        if (copy_from_user(&value,
                           (int __user *)arg,
                           sizeof(value))) {
            ret = -EFAULT;
            break;
        }

        if (mutex_lock_interruptible(&device_mutex))
            return -ERESTARTSYS;

        device_value = value;

        mutex_unlock(&device_mutex);

        pr_info("myioctl: Value set to %d\n", value);
        break;

    case MYDEV_EXCHANGE:

        if (copy_from_user(&value,
                           (int __user *)arg,
                           sizeof(value))) {
            ret = -EFAULT;
            break;
        }

        if (mutex_lock_interruptible(&device_mutex))
            return -ERESTARTSYS;

        device_value = value;
        value = device_value;

        mutex_unlock(&device_mutex);

        if (copy_to_user((int __user *)arg,
                         &value,
                         sizeof(value)))
            ret = -EFAULT;

        break;

    default:
        ret = -ENOTTY;
        break;
    }

    return ret;
}

/* Release callback */
static int my_release(struct inode *inode, struct file *file)
{
    pr_info("myioctl: Device released\n");
    return 0;
}

/* File operations */
static const struct file_operations my_fops = {
    .owner          = THIS_MODULE,
    .open           = my_open,
    .unlocked_ioctl = my_ioctl,
    .release        = my_release,
};

/* Module initialization */
static int __init ioctl_init(void)
{
    int ret;

    ret = alloc_chrdev_region(&dev_num, 0, 1, DEVICE_NAME);

    if (ret < 0)
        return ret;

    cdev_init(&my_cdev, &my_fops);
    my_cdev.owner = THIS_MODULE;

    ret = cdev_add(&my_cdev, dev_num, 1);

    if (ret < 0) {
        unregister_chrdev_region(dev_num, 1);
        return ret;
    }

    pr_info("myioctl: Registered Major=%d Minor=%d\n",
            MAJOR(dev_num), MINOR(dev_num));

    return 0;
}

/* Module cleanup */
static void __exit ioctl_exit(void)
{
    cdev_del(&my_cdev);
    unregister_chrdev_region(dev_num, 1);

    pr_info("myioctl: Driver removed\n");
}

module_init(ioctl_init);
module_exit(ioctl_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("NS Learning Hub");
MODULE_DESCRIPTION("IOCTL Based Character Device Driver");
MODULE_VERSION("1.0");

8. Understanding the Driver Code

8.1 unlocked_ioctl()

.unlocked_ioctl = my_ioctl

This connects the user-space ioctl() system call to the driver's IOCTL callback.

8.2 Command Validation

if (_IOC_TYPE(cmd) != MYDEV_MAGIC)
    return -ENOTTY;

if (_IOC_NR(cmd) > 3)
    return -ENOTTY;

The driver checks the command type and command number before processing it.

8.3 Device Reset

MYDEV_RESET resets the stored value and restores the device status.

8.4 Reading Status

MYDEV_GET_STATUS copies the status from kernel memory to the user-space application.

8.5 Setting a Value

MYDEV_SET_VALUE receives an integer from the application and stores it in the driver.

8.6 Bidirectional Exchange

MYDEV_EXCHANGE receives an integer and returns the value after processing it. This example echoes the value to demonstrate bidirectional transfer.

Kernel compatibility: This example uses unlocked_ioctl for a normal 64-bit Linux environment. Drivers supporting 32-bit applications on a 64-bit kernel may also need a carefully implemented compat_ioctl handler.

11. Create the User-Space Test Application

Create the test application:

nano ioctl_test.c

File: ioctl_test.c

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/ioctl.h>
#include <stdlib.h>

#include "my_ioctl.h"

int main(void)
{
    int fd;
    int value;
    int status;

    fd = open("/dev/myioctl", O_RDWR);

    if (fd < 0) {
        perror("open");
        return 1;
    }

    printf("IOCTL Character Driver Test\n");

    /* Set device value */
    value = 125;

    if (ioctl(fd, MYDEV_SET_VALUE, &value) < 0) {
        perror("MYDEV_SET_VALUE");
        close(fd);
        return 1;
    }

    printf("Value sent to driver: %d\n", value);

    /* Exchange value */
    value = 250;

    if (ioctl(fd, MYDEV_EXCHANGE, &value) < 0) {
        perror("MYDEV_EXCHANGE");
        close(fd);
        return 1;
    }

    printf("Value returned by driver: %d\n", value);

    /* Get device status */
    if (ioctl(fd, MYDEV_GET_STATUS, &status) < 0) {
        perror("MYDEV_GET_STATUS");
        close(fd);
        return 1;
    }

    printf("Device status: %d\n", status);

    /* Reset device */
    if (ioctl(fd, MYDEV_RESET) < 0) {
        perror("MYDEV_RESET");
        close(fd);
        return 1;
    }

    printf("Device reset command completed\n");

    close(fd);

    return 0;
}

Suppose the kernel reports major number 240. Create the device node:

sudo mknod /dev/myioctl c 240 0
sudo chmod 666 /dev/myioctl

ls -l /dev/myioctl

Replace 240 with the actual major number reported by your system.

19. Lesson Summary

  • IOCTL provides device-specific control operations.
  • The user-space ioctl() call communicates with the driver's unlocked_ioctl callback.
  • _IO, _IOR, _IOW and _IOWR define common IOCTL command formats.
  • copy_from_user() and copy_to_user() safely transfer data across the user/kernel boundary.
  • A shared header file keeps kernel and user-space command definitions consistent.
  • Command validation and input validation are essential for reliable driver design.
