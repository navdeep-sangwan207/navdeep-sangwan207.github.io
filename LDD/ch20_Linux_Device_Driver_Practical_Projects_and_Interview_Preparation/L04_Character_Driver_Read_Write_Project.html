1. Introduction

A character driver allows user-space applications to communicate with a kernel device through standard file operations.

Linux provides the struct file_operations structure to connect system calls such as open(), read(), write() and close() to driver callback functions.

In this project, we will develop a complete memory-backed character driver. It will accept data from a user-space application, store it in a kernel buffer and return that data when the application reads the device.

2. Project Objectives

  • Register a character device dynamically.
  • Create a device node in /dev.
  • Implement open and release callbacks.
  • Implement read and write callbacks.
  • Transfer data using copy_to_user() and copy_from_user().
  • Manage file position using loff_t.
  • Test the driver using user-space applications.
  • Handle invalid input and buffer boundaries.

4. Understanding struct file_operations

The struct file_operations structure contains function pointers that define how a device responds to file operations.

static const struct file_operations my_fops = {
    .owner   = THIS_MODULE,
    .open    = my_open,
    .read    = my_read,
    .write   = my_write,
    .release = my_release,
};
CallbackPurpose
open()Called when a process opens the device file.
read()Transfers data from the kernel to user space.
write()Transfers data from user space to the kernel.
release()Called when the open file description is released.

5. Understanding copy_to_user()

copy_to_user() transfers data from kernel memory to user-space memory.

unsigned long copy_to_user(
    void __user *to,
    const void *from,
    unsigned long count
);

Example:

char message[] = "Hello from kernel";

if (copy_to_user(user_buffer, message, sizeof(message)))
    return -EFAULT;

The function returns the number of bytes that could not be copied. A return value of zero indicates successful transfer.

6. Understanding copy_from_user()

copy_from_user() transfers data from user-space memory into kernel memory.

unsigned long copy_from_user(
    void *to,
    const void __user *from,
    unsigned long count
);

Example:

char kernel_buffer[100];

if (copy_from_user(kernel_buffer, user_buffer, count))
    return -EFAULT;

The kernel must validate input sizes and never assume that user-space pointers are valid.

8. Complete Character Driver Source Code

Create the driver source file:

nano char_rw.c

File: char_rw.c

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/uaccess.h>
#include <linux/mutex.h>
#include <linux/errno.h>

#define DEVICE_NAME "mychar"
#define BUFFER_SIZE 1024

static dev_t dev_num;
static struct cdev my_cdev;

static char device_buffer[BUFFER_SIZE];
static size_t data_size;
static DEFINE_MUTEX(buffer_mutex);

/* Open callback */
static int my_open(struct inode *inode, struct file *file)
{
    pr_info("mychar: Device opened\n");
    return 0;
}

/* Read callback */
static ssize_t my_read(
    struct file *file,
    char __user *user_buffer,
    size_t count,
    loff_t *offset)
{
    size_t bytes_to_read;
    size_t remaining;

    if (mutex_lock_interruptible(&buffer_mutex))
        return -ERESTARTSYS;

    if (*offset >= data_size) {
        mutex_unlock(&buffer_mutex);
        return 0;
    }

    remaining = data_size - *offset;
    bytes_to_read = min(count, remaining);

    if (copy_to_user(user_buffer,
                     device_buffer + *offset,
                     bytes_to_read)) {
        mutex_unlock(&buffer_mutex);
        return -EFAULT;
    }

    *offset += bytes_to_read;

    mutex_unlock(&buffer_mutex);

    pr_info("mychar: Read %zu bytes\n", bytes_to_read);

    return bytes_to_read;
}

/* Write callback */
static ssize_t my_write(
    struct file *file,
    const char __user *user_buffer,
    size_t count,
    loff_t *offset)
{
    size_t bytes_to_write;

    if (mutex_lock_interruptible(&buffer_mutex))
        return -ERESTARTSYS;

    if (*offset < 0 || *offset >= BUFFER_SIZE) {
        mutex_unlock(&buffer_mutex);
        return -ENOSPC;
    }

    bytes_to_write = min(count,
                         (size_t)(BUFFER_SIZE - *offset));

    if (copy_from_user(device_buffer + *offset,
                       user_buffer,
                       bytes_to_write)) {
        mutex_unlock(&buffer_mutex);
        return -EFAULT;
    }

    *offset += bytes_to_write;

    if (*offset > data_size)
        data_size = *offset;

    mutex_unlock(&buffer_mutex);

    pr_info("mychar: Wrote %zu bytes\n", bytes_to_write);

    return bytes_to_write;
}

/* Release callback */
static int my_release(struct inode *inode, struct file *file)
{
    pr_info("mychar: Device released\n");
    return 0;
}

/* File operations */
static const struct file_operations my_fops = {
    .owner   = THIS_MODULE,
    .open    = my_open,
    .read    = my_read,
    .write   = my_write,
    .release = my_release,
};

/* Module initialization */
static int __init char_rw_init(void)
{
    int ret;

    ret = alloc_chrdev_region(&dev_num, 0, 1, DEVICE_NAME);
    if (ret < 0) {
        pr_err("mychar: Device number allocation failed\n");
        return ret;
    }

    cdev_init(&my_cdev, &my_fops);
    my_cdev.owner = THIS_MODULE;

    ret = cdev_add(&my_cdev, dev_num, 1);
    if (ret < 0) {
        pr_err("mychar: cdev_add failed\n");
        unregister_chrdev_region(dev_num, 1);
        return ret;
    }

    pr_info("mychar: Registered Major=%d Minor=%d\n",
            MAJOR(dev_num), MINOR(dev_num));

    return 0;
}

/* Module cleanup */
static void __exit char_rw_exit(void)
{
    cdev_del(&my_cdev);
    unregister_chrdev_region(dev_num, 1);

    pr_info("mychar: Driver removed\n");
}

module_init(char_rw_init);
module_exit(char_rw_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("NS Learning Hub");
MODULE_DESCRIPTION("Memory-Backed Character Read Write Driver");
MODULE_VERSION("1.0");

9. Code Explanation

9.1 Kernel Buffer

static char device_buffer[BUFFER_SIZE];
static size_t data_size;

The array stores data written by user-space applications. data_size records the number of valid bytes currently stored.

9.2 Mutex Protection

static DEFINE_MUTEX(buffer_mutex);

The mutex protects shared buffer data against concurrent access by multiple processes.

9.3 Read Operation

The read callback performs these steps:

  1. Acquire the mutex.
  2. Check whether the current file offset has reached the end of stored data.
  3. Calculate the number of bytes available.
  4. Copy data into the user-space buffer.
  5. Update the file offset.
  6. Release the mutex.
  7. Return the number of bytes successfully transferred.

9.4 Write Operation

The write callback:

  1. Acquires the mutex.
  2. Validates the file offset.
  3. Calculates available buffer space.
  4. Copies data from user space into kernel memory.
  5. Updates the file offset and valid data size.
  6. Releases the mutex.

9.5 File Offset

The loff_t *offset argument tracks the current position in the device's data stream. The read and write callbacks advance it after successful transfers.

Design limitation: This teaching driver uses a single shared memory buffer. Different open file descriptions have independent offsets, but all processes share the same data. The project does not implement llseek(), device reset or per-open buffers.

12. Create the Device Node

The driver registration does not automatically create a device file. For this practical, create one manually using mknod.

First, identify the major number from the kernel log or:

cat /proc/devices | grep mychar

Suppose the major number is 240 and the minor number is 0. Create the node:

sudo mknod /dev/mychar c 240 0

sudo chmod 666 /dev/mychar

ls -l /dev/mychar
Important: Replace 240 with the actual major number allocated on your machine. The permissions above are only for a controlled learning VM. Production devices should use appropriate ownership and permissions.

13. Create a User-Space Test Application

Create the test program:

nano test_char.c

File: test_char.c

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>
#include <errno.h>

int main(void)
{
    int fd;
    char write_buffer[] = "Hello from user space!";
    char read_buffer[100] = {0};
    ssize_t bytes;

    fd = open("/dev/mychar", O_RDWR);

    if (fd < 0) {
        perror("open");
        return 1;
    }

    bytes = write(fd, write_buffer, strlen(write_buffer));

    if (bytes < 0) {
        perror("write");
        close(fd);
        return 1;
    }

    printf("Written: %zd bytes\n", bytes);

    /* Reset file offset to read from the beginning */
    if (lseek(fd, 0, SEEK_SET) < 0) {
        perror("lseek");
        close(fd);
        return 1;
    }

    bytes = read(fd, read_buffer, sizeof(read_buffer) - 1);

    if (bytes < 0) {
        perror("read");
        close(fd);
        return 1;
    }

    read_buffer[bytes] = '\0';

    printf("Read data: %s\n", read_buffer);

    close(fd);

    return 0;
}

This program opens the device, writes a message, resets the file position, reads the stored data and closes the device.

19. Lesson Summary

  • A character driver connects user-space file operations with kernel callbacks.
  • struct file_operations defines the driver's supported operations.
  • copy_to_user() transfers data from kernel to user space.
  • copy_from_user() transfers data from user space to kernel.
  • loff_t tracks the current file position.
  • Mutexes protect shared driver data from concurrent access.
  • mknod creates a device node after the driver has registered its device number.
  • A user-space C program can test the driver through open(), read(), write() and close().
