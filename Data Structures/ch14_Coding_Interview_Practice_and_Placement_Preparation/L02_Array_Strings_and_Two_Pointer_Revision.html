3. Find Maximum and Minimum Elements

Scan the array once while maintaining the current minimum and maximum.

Complete C11 Program

#include <stdio.h> #define MAX 100 int main(void) { int arr[MAX]; int n, i; int minimum, maximum; printf("Enter number of elements: "); if (scanf("%d", &n) != 1 || n <= 0 || n > MAX) return 1; printf("Enter elements:\n"); for (i = 0; i < n; i++) { if (scanf("%d", &arr[i]) != 1) return 1; } minimum = maximum = arr[0]; for (i = 1; i < n; i++) { if (arr[i] < minimum) minimum = arr[i]; if (arr[i] > maximum) maximum = arr[i]; } printf("Minimum = %d\n", minimum); printf("Maximum = %d\n", maximum); return 0; }

Expected Output

Enter number of elements: 6 Enter elements: 12 5 28 3 19 8 Minimum = 3 Maximum = 28

Time: O(n). Auxiliary space: O(1).

4. Reverse an Array Using Two Pointers

Place one pointer at the beginning and another at the end. Swap the elements and move both pointers toward the center.

Complete C11 Program

#include <stdio.h> #define MAX 100 void reverseArray(int arr[], int n) { int left = 0; int right = n - 1; while (left < right) { int temp = arr[left]; arr[left] = arr[right]; arr[right] = temp; left++; right--; } } int main(void) { int arr[] = {10, 20, 30, 40, 50}; int n = 5; int i; reverseArray(arr, n); printf("Reversed array: "); for (i = 0; i < n; i++) printf("%d ", arr[i]); printf("\n"); return 0; }

Expected Output

Reversed array: 50 40 30 20 10

Time: O(n). Auxiliary space: O(1).

5. Move All Zeros to the End

Move all zero values to the end while preserving the relative order of non-zero elements.

Approach: Write Pointer

  1. Maintain a position for the next non-zero element.
  2. Scan the array from left to right.
  3. Copy each non-zero element to the next write position.
  4. Fill the remaining positions with zeros.

Complete C11 Program

#include <stdio.h> void moveZeros(int arr[], int n) { int write = 0; int i; for (i = 0; i < n; i++) { if (arr[i] != 0) arr[write++] = arr[i]; } while (write < n) arr[write++] = 0; } int main(void) { int arr[] = {0, 1, 0, 3, 12}; int n = 5; int i; moveZeros(arr, n); printf("Result: "); for (i = 0; i < n; i++) printf("%d ", arr[i]); printf("\n"); return 0; }

Expected Output

Result: 1 3 12 0 0

Time: O(n). Auxiliary space: O(1).

6. Remove Duplicates from a Sorted Array

Given a sorted array, remove duplicate values in-place and return the number of unique elements.

Complete C11 Program

#include <stdio.h> int removeDuplicates(int arr[], int n) { int uniqueCount; int i; if (n == 0) return 0; uniqueCount = 1; for (i = 1; i < n; i++) { if (arr[i] != arr[uniqueCount - 1]) { arr[uniqueCount] = arr[i]; uniqueCount++; } } return uniqueCount; } int main(void) { int arr[] = {1, 1, 2, 2, 3, 4, 4, 5}; int n = 8; int k, i; k = removeDuplicates(arr, n); printf("Unique count = %d\n", k); printf("Unique elements: "); for (i = 0; i < k; i++) printf("%d ", arr[i]); printf("\n"); return 0; }

Expected Output

Unique count = 5 Unique elements: 1 2 3 4 5

Time: O(n). Auxiliary space: O(1).

The input must be sorted. Values beyond the returned unique count are not part of the resulting logical array.

7. Rotate an Array to the Right by K Positions

The reversal algorithm rotates an array in-place using three reversals.

Example

Original: 1 2 3 4 5 6 7 K = 3 Result: 5 6 7 1 2 3 4

Complete C11 Program

#include <stdio.h> void reverse(int arr[], int left, int right) { while (left < right) { int temp = arr[left]; arr[left] = arr[right]; arr[right] = temp; left++; right--; } } void rotateRight(int arr[], int n, int k) { if (n <= 1) return; k %= n; if (k < 0) k += n; if (k == 0) return; reverse(arr, 0, n - 1); reverse(arr, 0, k - 1); reverse(arr, k, n - 1); } int main(void) { int arr[] = {1, 2, 3, 4, 5, 6, 7}; int n = 7; int k = 3; int i; rotateRight(arr, n, k); printf("Rotated array: "); for (i = 0; i < n; i++) printf("%d ", arr[i]); printf("\n"); return 0; }

Expected Output

Rotated array: 5 6 7 1 2 3 4

Time: O(n). Auxiliary space: O(1).

8. Maximum Subarray Sum – Kadane's Algorithm

Find the maximum sum of a non-empty contiguous subarray.

Core Idea

At each position, either extend the previous subarray or start a new subarray from the current element.

currentSum = max(arr[i], currentSum + arr[i]) maximumSum = max(maximumSum, currentSum)

Complete C11 Program

#include <stdio.h> #define MAX 100 int main(void) { int arr[MAX]; int n, i; long long currentSum; long long maximumSum; printf("Enter number of elements: "); if (scanf("%d", &n) != 1 || n <= 0 || n > MAX) return 1; printf("Enter elements:\n"); for (i = 0; i < n; i++) { if (scanf("%d", &arr[i]) != 1) return 1; } currentSum = maximumSum = arr[0]; for (i = 1; i < n; i++) { long long extend = currentSum + arr[i]; if (arr[i] > extend) currentSum = arr[i]; else currentSum = extend; if (currentSum > maximumSum) maximumSum = currentSum; } printf("Maximum subarray sum = %lld\n", maximumSum); return 0; }

Expected Output

Enter number of elements: 9 Enter elements: -2 1 -3 4 -1 2 1 -5 4 Maximum subarray sum = 6

The maximum-sum subarray is [4, -1, 2, 1].

Time: O(n). Auxiliary space: O(1).

9. Two Sum in an Unsorted Array Using Sorting and Two Pointers

For an unsorted array, one option is to sort the elements and then apply two pointers. This version returns the values, not the original indices.

Complete C11 Program

#include <stdio.h> #include <stdlib.h> int compareInt(const void *a, const void *b) { int x = *(const int *)a; int y = *(const int *)b; return (x > y) - (x < y); } int main(void) { int arr[] = {8, 3, 11, 2, 7, 5}; int n = 6; int target = 10; int left = 0; int right = n - 1; int found = 0; qsort(arr, n, sizeof(arr[0]), compareInt); while (left < right) { long long sum = (long long)arr[left] + arr[right]; if (sum == target) { printf("Pair: %d %d\n", arr[left], arr[right]); found = 1; break; } else if (sum < target) { left++; } else { right--; } } if (!found) printf("No pair found.\n"); return 0; }

Expected Output

Pair: 2 8

Time: O(n log n) for sorting plus O(n) for scanning. Auxiliary space: qsort implementation-dependent.

If the question asks for original array indices, sorting the values directly loses their original positions. Store each value with its original index or use a hash table.

10. Merge Two Sorted Arrays

Use two pointers to merge two sorted arrays into a third sorted array.

Complete C11 Program

#include <stdio.h> #define MAX 100 int main(void) { int a[] = {1, 3, 5, 7}; int b[] = {2, 4, 6, 8}; int m = 4, n = 4; int result[2 * MAX]; int i = 0, j = 0, k = 0; while (i < m && j < n) { if (a[i] <= b[j]) result[k++] = a[i++]; else result[k++] = b[j++]; } while (i < m) result[k++] = a[i++]; while (j < n) result[k++] = b[j++]; printf("Merged array: "); for (i = 0; i < k; i++) printf("%d ", result[i]); printf("\n"); return 0; }

Expected Output

Merged array: 1 2 3 4 5 6 7 8

Time: O(m+n). Auxiliary space: O(m+n) for the output array.

11. String Reversal

A C string is a character array terminated by the null character '\0'. Two pointers can reverse a string in-place.

Complete C11 Program

#include <stdio.h> #include <string.h> void reverseString(char str[]) { size_t left = 0; size_t right = strlen(str); if (right == 0) return; right--; while (left < right) { char temp = str[left]; str[left] = str[right]; str[right] = temp; left++; right--; } } int main(void) { char str[] = "placement"; reverseString(str); printf("Reversed string: %s\n", str); return 0; }

Expected Output

Reversed string: tnemecalp

Time: O(n). Auxiliary space: O(1).

12. Check Whether a String Is a Palindrome

A palindrome reads the same forward and backward. This implementation compares characters from both ends.

Complete C11 Program

#include <stdio.h> #include <string.h> int isPalindrome(const char str[]) { size_t left = 0; size_t right = strlen(str); if (right == 0) return 1; right--; while (left < right) { if (str[left] != str[right]) return 0; left++; right--; } return 1; } int main(void) { char str[] = "madam"; if (isPalindrome(str)) printf("Palindrome\n"); else printf("Not a palindrome\n"); return 0; }

Expected Output

Palindrome

Time: O(n). Auxiliary space: O(1).

This version is case-sensitive and considers spaces and punctuation. For example, "Madam" is not a palindrome under these exact comparison rules.

13. Check Whether Two Strings Are Anagrams

Two strings are anagrams if they contain the same characters with the same frequencies, regardless of order.

Complete C11 Program – English Letters, Case-Insensitive

#include <stdio.h> #include <string.h> #include <ctype.h> int areAnagrams(const char a[], const char b[]) { int frequency[26] = {0}; size_t i; if (strlen(a) != strlen(b)) return 0; for (i = 0; a[i] != '\0'; i++) { unsigned char ch = (unsigned char)a[i]; if (!isalpha(ch)) return 0; frequency[tolower(ch) - 'a']++; } for (i = 0; b[i] != '\0'; i++) { unsigned char ch = (unsigned char)b[i]; if (!isalpha(ch)) return 0; frequency[tolower(ch) - 'a']--; } for (i = 0; i < 26; i++) { if (frequency[i] != 0) return 0; } return 1; } int main(void) { char a[] = "Listen"; char b[] = "Silent"; if (areAnagrams(a, b)) printf("Anagrams\n"); else printf("Not anagrams\n"); return 0; }

Expected Output

Anagrams

Time: O(n). Auxiliary space: O(1), because the frequency array has a fixed size of 26.

This implementation accepts English alphabetic characters only. It rejects spaces, punctuation and non-English letters. Unicode anagram comparison requires a different character-processing approach.

14. First Non-Repeating Character

Find the first character that occurs exactly once in a string.

Complete C11 Program – ASCII Input

#include <stdio.h> #include <limits.h> int main(void) { char str[] = "swiss"; int frequency[UCHAR_MAX + 1] = {0}; int i; for (i = 0; str[i] != '\0'; i++) { unsigned char ch = (unsigned char)str[i]; frequency[ch]++; } for (i = 0; str[i] != '\0'; i++) { unsigned char ch = (unsigned char)str[i]; if (frequency[ch] == 1) { printf("First non-repeating character = %c\n", str[i]); return 0; } } printf("No non-repeating character found.\n"); return 0; }

Expected Output

First non-repeating character = w

Time: O(n + R), where R is the number of possible unsigned-char values. Auxiliary space: O(R).

15. Longest Substring Without Repeating Characters

Find the length of the longest substring containing no repeated bytes. A sliding window and last-seen index array solve this efficiently.

Complete C11 Program – ASCII/Byte Strings

#include <stdio.h> #include <limits.h> #include <string.h> int longestUniqueSubstring(const char str[]) { int lastSeen[UCHAR_MAX + 1]; int left = 0; int best = 0; int i; for (i = 0; i <= UCHAR_MAX; i++) lastSeen[i] = -1; for (i = 0; str[i] != '\0'; i++) { unsigned char ch = (unsigned char)str[i]; if (lastSeen[ch] >= left) left = lastSeen[ch] + 1; lastSeen[ch] = i; if (i - left + 1 > best) best = i - left + 1; } return best; } int main(void) { char str[] = "abcabcbb"; printf("Longest unique substring length = %d\n", longestUniqueSubstring(str)); return 0; }

Expected Output

Longest unique substring length = 3

One longest substring is "abc".

Time: O(n + R) initialization plus O(n) scanning. Auxiliary space: O(R), where R is the unsigned-char range.

16. Longest Common Prefix

Find the longest prefix shared by all strings in an array.

Complete C11 Program

#include <stdio.h> #include <string.h> #define COUNT 4 void longestCommonPrefix( const char *words[], char result[], size_t capacity) { size_t i = 0; size_t j; if (capacity == 0) return; result[0] = '\0'; while (words[0][i] != '\0') { char ch = words[0][i]; for (j = 1; j < COUNT; j++) { if (words[j][i] != ch) return; } if (i + 1 >= capacity) return; result[i] = ch; i++; result[i] = '\0'; } } int main(void) { const char *words[COUNT] = { "flower", "flow", "flight", "flute" }; char prefix[100]; longestCommonPrefix(words, prefix, sizeof(prefix)); if (prefix[0] == '\0') printf("No common prefix.\n"); else printf("Longest common prefix = %s\n", prefix); return 0; }

Expected Output

Longest common prefix = fl

Time: O(S), where S is the total number of characters examined. Auxiliary space: O(1), excluding the output buffer.

19. Common Interview Mistakes

  • Accessing arr[0] without checking whether n is positive.
  • Using two pointers on unsorted data without a valid ordering argument.
  • Forgetting to normalize k when rotating an array.
  • Assuming all values are positive in maximum-subarray problems.
  • Confusing subarray and subsequence.
  • Ignoring integer overflow in sums.
  • Using a fixed frequency table for an unrestricted value range.
  • Comparing strings without defining case and whitespace rules.
  • Forgetting the null terminator when modifying C strings.
  • Returning an answer for an empty or invalid input without defining behavior.

20. Interview Questions and Answers

Q1. What is the difference between an array and a linked list?

Arrays store elements contiguously and support O(1) indexed access. Linked lists store nodes connected by pointers and require traversal for positional access.

Q2. Why is array access O(1)?

The address of an element can be calculated directly from the base address, index and element size.

Q3. What is the two-pointer technique?

It uses two positions that move according to a problem-specific invariant to reduce repeated work.

Q4. Why does Kadane's algorithm work?

At each index, an optimal maximum-sum subarray ending there either starts at the current element or extends the best subarray ending at the previous index.

Q5. Can binary search be used on an unsorted array?

Not with the ordinary sorted-array method. Binary search requires an ordered or monotonic search condition.

Q6. What is an in-place algorithm?

An algorithm that modifies the input directly and uses only a limited amount of additional storage, often O(1) auxiliary space.

Q7. What is the difference between a substring and a subsequence?

A substring is contiguous. A subsequence preserves relative order but may skip characters.

Q8. How can two strings be checked for anagrams?

Compare their character frequencies after applying the exact normalization rules required by the question.

Q9. What is the sliding window technique?

It maintains a contiguous range and updates its state as the range moves through the input.

Q10. What is Kadane's algorithm complexity?

O(n) time and O(1) auxiliary space.
