9. Final Coding Assessment – Problem 1

Problem: Find the second largest distinct element in an integer array. Return a status indicating whether such an element exists.

Complete C11 Program

#include <stdio.h> #include <limits.h> int secondLargest(const int arr[], int n, int *result) { if (arr == NULL || result == NULL || n < 2) return 0; int largest = INT_MIN; int second = INT_MIN; int hasLargest = 0; int hasSecond = 0; for (int i = 0; i < n; i++) { if (!hasLargest || arr[i] > largest) { if (hasLargest) { second = largest; hasSecond = 1; } largest = arr[i]; hasLargest = 1; } else if (arr[i] < largest) { if (!hasSecond || arr[i] > second) { second = arr[i]; hasSecond = 1; } } } if (!hasSecond) return 0; *result = second; return 1; } int main(void) { int arr[] = {12, 35, 1, 10, 34, 1}; int n = sizeof(arr) / sizeof(arr[0]); int answer; if (secondLargest(arr, n, &answer)) printf("Second largest = %d\n", answer); else printf("Second largest does not exist\n"); return 0; }
Second largest = 34

Time: O(n). Auxiliary space: O(1).

10. Final Coding Assessment – Problem 2

Problem: Determine whether an array contains a duplicate value. Use sorting followed by adjacent comparison.

Complete C11 Program

#include <stdio.h> #include <stdlib.h> int compareInt(const void *a, const void *b) { int x = *(const int *)a; int y = *(const int *)b; return (x > y) - (x < y); } int containsDuplicate(int arr[], int n) { if (arr == NULL || n < 2) return 0; qsort(arr, (size_t)n, sizeof(arr[0]), compareInt); for (int i = 1; i < n; i++) { if (arr[i] == arr[i - 1]) return 1; } return 0; } int main(void) { int arr[] = {4, 2, 7, 2, 9}; int n = sizeof(arr) / sizeof(arr[0]); printf("%s\n", containsDuplicate(arr, n) ? "Duplicate found" : "No duplicate"); return 0; }
Duplicate found

Time: O(n log n) for sorting plus O(n) scanning. Auxiliary space: Implementation-dependent for qsort; the scan itself uses O(1).

This method changes the original array order. If the original order must be preserved, copy the array before sorting or use a hash set.

11. Final Coding Assessment – Problem 3

Problem: Given a sorted array, remove duplicates in place and return the number of distinct elements.

Complete C11 Program

#include <stdio.h> int removeDuplicates(int arr[], int n) { if (arr == NULL || n <= 0) return 0; int write = 1; for (int read = 1; read < n; read++) { if (arr[read] != arr[write - 1]) { arr[write] = arr[read]; write++; } } return write; } int main(void) { int arr[] = {1, 1, 2, 2, 2, 3, 4, 4}; int n = sizeof(arr) / sizeof(arr[0]); int newLength = removeDuplicates(arr, n); printf("Distinct count = %d\n", newLength); printf("Array: "); for (int i = 0; i < newLength; i++) printf("%d ", arr[i]); printf("\n"); return 0; }
Distinct count = 4 Array: 1 2 3 4

Time: O(n). Auxiliary space: O(1).

12. Final Coding Assessment – Problem 4

Problem: Find the minimum number of jumps required to reach the final array index. Each element represents the maximum jump length from that position. Return -1 if the end cannot be reached.

Complete C11 Program – Greedy Approach

#include <stdio.h> int minimumJumps(const int arr[], int n) { if (arr == NULL || n <= 0) return -1; if (n == 1) return 0; if (arr[0] <= 0) return -1; int jumps = 1; int currentEnd = arr[0]; int farthest = arr[0]; for (int i = 1; i < n - 1; i++) { if (i > farthest) return -1; if ((long long)i + arr[i] > farthest) { long long reach = (long long)i + arr[i]; farthest = reach > n - 1 ? n - 1 : (int)reach; } if (i == currentEnd) { if (farthest <= currentEnd) return -1; jumps++; currentEnd = farthest; if (currentEnd >= n - 1) return jumps; } } return currentEnd >= n - 1 ? jumps : -1; } int main(void) { int arr[] = {2, 3, 1, 1, 4}; int n = sizeof(arr) / sizeof(arr[0]); printf("Minimum jumps = %d\n", minimumJumps(arr, n)); return 0; }
Minimum jumps = 2

Time: O(n). Auxiliary space: O(1).
