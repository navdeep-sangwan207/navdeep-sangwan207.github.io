2. Reverse a String

Problem: Reverse a given string without using a second character array.

#include <stdio.h>
#include <string.h>

void reverse_string(char str[])
{
    int left = 0;
    int right = (int)strlen(str) - 1;

    while (left < right)
    {
        char temp = str[left];
        str[left] = str[right];
        str[right] = temp;

        left++;
        right--;
    }
}

int main(void)
{
    char str[] = "HELLO";

    reverse_string(str);

    printf("Reversed string: %s\n", str);

    return 0;
}
Expected Output: 
Reversed string: OLLEH

Time: O(n). Auxiliary Space: O(1).

Technique: Two pointers.

3. Check Whether a String Is a Palindrome

Problem: Determine whether a string reads the same forward and backward.

#include <stdio.h>
#include <string.h>

int is_palindrome(const char str[])
{
    int left = 0;
    int right = (int)strlen(str) - 1;

    while (left < right)
    {
        if (str[left] != str[right])
            return 0;

        left++;
        right--;
    }

    return 1;
}

int main(void)
{
    char str[] = "MADAM";

    if (is_palindrome(str))
        printf("Palindrome\n");
    else
        printf("Not a palindrome\n");

    return 0;
}
Expected Output: 
Palindrome

Time: O(n). Auxiliary Space: O(1).

4. Find the First Non-Repeating Character

Problem: Find the first character that occurs exactly once.

#include <stdio.h>

int main(void)
{
    char str[] = "swiss";
    int frequency[256] = {0};

    for (int i = 0; str[i] != '\0'; i++)
        frequency[(unsigned char)str[i]]++;

    int found = 0;

    for (int i = 0; str[i] != '\0'; i++)
    {
        if (frequency[(unsigned char)str[i]] == 1)
        {
            printf("First non-repeating character: %c\n", str[i]);
            found = 1;
            break;
        }
    }

    if (!found)
        printf("No non-repeating character\n");

    return 0;
}
Expected Output: 
First non-repeating character: w

Time: O(n). Auxiliary Space: O(1) for a fixed 256-entry frequency array.

5. Check Whether Two Strings Are Anagrams

Problem: Two strings are anagrams if they contain the same characters with the same frequencies.

#include <stdio.h>
#include <string.h>

int are_anagrams(const char a[], const char b[])
{
    int frequency[256] = {0};

    if (strlen(a) != strlen(b))
        return 0;

    for (int i = 0; a[i] != '\0'; i++)
    {
        frequency[(unsigned char)a[i]]++;
        frequency[(unsigned char)b[i]]--;
    }

    for (int i = 0; i < 256; i++)
    {
        if (frequency[i] != 0)
            return 0;
    }

    return 1;
}

int main(void)
{
    char a[] = "listen";
    char b[] = "silent";

    if (are_anagrams(a, b))
        printf("Anagrams\n");
    else
        printf("Not anagrams\n");

    return 0;
}
Expected Output: 
Anagrams

Time: O(n + 256), effectively O(n). Auxiliary Space: O(1) for the fixed frequency array.

6. Remove Duplicate Characters

Problem: Remove repeated characters while preserving the first occurrence order.

#include <stdio.h>

void remove_duplicates(char str[])
{
    int seen[256] = {0};
    int read = 0;
    int write = 0;

    while (str[read] != '\0')
    {
        unsigned char ch = (unsigned char)str[read];

        if (!seen[ch])
        {
            seen[ch] = 1;
            str[write++] = str[read];
        }

        read++;
    }

    str[write] = '\0';
}

int main(void)
{
    char str[] = "programming";

    remove_duplicates(str);

    printf("Result: %s\n", str);

    return 0;
}
Expected Output: 
Result: progamin

Time: O(n). Auxiliary Space: O(1) for a fixed-size character table.

7. Find the Maximum Occurring Character

Problem: Find the character with the highest frequency. If there is a tie, return the first such character in the string.

#include <stdio.h>

int main(void)
{
    char str[] = "success";
    int frequency[256] = {0};

    for (int i = 0; str[i] != '\0'; i++)
        frequency[(unsigned char)str[i]]++;

    int max_frequency = 0;
    unsigned char result = '\0';

    for (int i = 0; str[i] != '\0'; i++)
    {
        unsigned char ch = (unsigned char)str[i];

        if (frequency[ch] > max_frequency)
        {
            max_frequency = frequency[ch];
            result = ch;
        }
    }

    if (result != '\0')
        printf("Maximum occurring character: %c\n", result);

    return 0;
}
Expected Output: 
Maximum occurring character: s

Time: O(n). Auxiliary Space: O(1) for the fixed character table.

8. Find the Longest Word in a Sentence

Problem: Find the longest sequence of non-space characters in a sentence.

#include <stdio.h>
#include <string.h>

int main(void)
{
    char str[] = "C programming is very useful";

    int max_start = 0;
    int max_length = 0;
    int current_start = 0;
    int current_length = 0;

    for (int i = 0; ; i++)
    {
        if (str[i] != ' ' && str[i] != '\0')
        {
            if (current_length == 0)
                current_start = i;

            current_length++;
        }
        else
        {
            if (current_length > max_length)
            {
                max_length = current_length;
                max_start = current_start;
            }

            current_length = 0;

            if (str[i] == '\0')
                break;
        }
    }

    printf("Longest word: ");

    for (int i = 0; i < max_length; i++)
        putchar(str[max_start + i]);

    printf("\nLength: %d\n", max_length);

    return 0;
}
Expected Output: 
Longest word: programming
Length: 11

Time: O(n). Auxiliary Space: O(1).

9. Check Whether One String Is a Rotation of Another

Problem: Check whether one string can be obtained by rotating another string.

Example:

String 1: ABCD
String 2: CDAB

Result: Rotation
#include <stdio.h>
#include <string.h>

int is_rotation(const char a[], const char b[])
{
    size_t n = strlen(a);

    if (n != strlen(b))
        return 0;

    if (n == 0)
        return 1;

    char doubled[2 * n + 1];

    strcpy(doubled, a);
    strcat(doubled, a);

    return strstr(doubled, b) != NULL;
}

int main(void)
{
    char a[] = "ABCD";
    char b[] = "CDAB";

    if (is_rotation(a, b))
        printf("Rotation\n");
    else
        printf("Not a rotation\n");

    return 0;
}
Expected Output: 
Rotation

Time: O(n²) worst case with the standard strstr implementation. Auxiliary Space: O(n).

The key idea is that if B is a rotation of A, then B must be a substring of A concatenated with itself.

10. Find the Longest Common Prefix

Problem: Given an array of strings, find the longest prefix common to all strings.

#include <stdio.h>
#include <string.h>

void longest_common_prefix(char *str[], int count)
{
    if (count == 0)
    {
        printf("No strings\n");
        return;
    }

    size_t length = strlen(str[0]);

    for (size_t i = 0; i < length; i++)
    {
        char ch = str[0][i];

        for (int j = 1; j < count; j++)
        {
            if (str[j][i] == '\0' || str[j][i] != ch)
            {
                printf("Longest common prefix: ");

                for (size_t k = 0; k < i; k++)
                    putchar(str[0][k]);

                printf("\n");
                return;
            }
        }
    }

    printf("Longest common prefix: %s\n", str[0]);
}

int main(void)
{
    char *words[] = {
        "flower",
        "flow",
        "flight"
    };

    longest_common_prefix(words, 3);

    return 0;
}
Expected Output: 
Longest common prefix: fl

Time: O(S), where S is the total number of characters examined. Auxiliary Space: O(1).

11. Check Whether a String Contains Only Digits

Problem: Determine whether every character in a non-empty string is a decimal digit.

#include <stdio.h>

int contains_only_digits(const char str[])
{
    if (str[0] == '\0')
        return 0;

    for (int i = 0; str[i] != '\0'; i++)
    {
        if (str[i] < '0' || str[i] > '9')
            return 0;
    }

    return 1;
}

int main(void)
{
    char str[] = "123456";

    if (contains_only_digits(str))
        printf("Only digits\n");
    else
        printf("Contains non-digit characters\n");

    return 0;
}
Expected Output: 
Only digits

Time: O(n). Auxiliary Space: O(1).

12. Longest Substring Without Repeating Characters

Problem: Find the length of the longest substring containing no repeated characters.

#include <stdio.h>
#include <string.h>

int longest_unique_substring(const char str[])
{
    int last_seen[256];

    for (int i = 0; i < 256; i++)
        last_seen[i] = -1;

    int left = 0;
    int max_length = 0;

    for (int right = 0; str[right] != '\0'; right++)
    {
        unsigned char ch = (unsigned char)str[right];

        if (last_seen[ch] >= left)
            left = last_seen[ch] + 1;

        last_seen[ch] = right;

        int length = right - left + 1;

        if (length > max_length)
            max_length = length;
    }

    return max_length;
}

int main(void)
{
    char str[] = "abcabcbb";

    printf("Longest length: %d\n",
           longest_unique_substring(str));

    return 0;
}
Expected Output: 
Longest length: 3

Time: O(n). Auxiliary Space: O(1) for a fixed 256-entry array.

Technique: Sliding window + last-seen index.

13. Check Whether a String Is a Subsequence of Another

Problem: Determine whether all characters of a smaller string occur in the same order in a larger string. They do not need to be adjacent.

#include <stdio.h>

int is_subsequence(const char small[], const char large[])
{
    int i = 0;
    int j = 0;

    while (small[i] != '\0' && large[j] != '\0')
    {
        if (small[i] == large[j])
            i++;

        j++;
    }

    return small[i] == '\0';
}

int main(void)
{
    char small[] = "ace";
    char large[] = "abcde";

    if (is_subsequence(small, large))
        printf("Subsequence\n");
    else
        printf("Not a subsequence\n");

    return 0;
}
Expected Output: 
Subsequence

Time: O(n + m). Auxiliary Space: O(1).

14. Find the Minimum Number of Character Deletions to Make Two Strings Anagrams

Problem: Find the minimum total number of character deletions required to make two strings anagrams.

#include <stdio.h>

int min_deletions_for_anagrams(const char a[], const char b[])
{
    int frequency[256] = {0};

    for (int i = 0; a[i] != '\0'; i++)
        frequency[(unsigned char)a[i]]++;

    for (int i = 0; b[i] != '\0'; i++)
        frequency[(unsigned char)b[i]]--;

    int deletions = 0;

    for (int i = 0; i < 256; i++)
    {
        int value = frequency[i];

        if (value < 0)
            value = -value;

        deletions += value;
    }

    return deletions;
}

int main(void)
{
    char a[] = "cde";
    char b[] = "abc";

    printf("Minimum deletions: %d\n",
           min_deletions_for_anagrams(a, b));

    return 0;
}
Expected Output: 
Minimum deletions: 4

Time: O(n + m + 256), effectively O(n + m). Auxiliary Space: O(1) for a fixed frequency array.

15. Count Occurrences of a Pattern in a String

Problem: Count all occurrences of a pattern, including overlapping matches.

#include <stdio.h>
#include <string.h>

int count_pattern(const char text[], const char pattern[])
{
    size_t n = strlen(text);
    size_t m = strlen(pattern);
    int count = 0;

    if (m == 0 || m > n)
        return 0;

    for (size_t i = 0; i <= n - m; i++)
    {
        if (memcmp(text + i, pattern, m) == 0)
            count++;
    }

    return count;
}

int main(void)
{
    char text[] = "AAAAA";
    char pattern[] = "AA";

    printf("Occurrences: %d\n",
           count_pattern(text, pattern));

    return 0;
}
Expected Output: 
Occurrences: 4

Time: O((n - m + 1)m) worst case. Auxiliary Space: O(1).

The matches occur at indices 0, 1, 2 and 3. This example demonstrates overlapping occurrences. KMP can solve this problem in O(n + m) time.

18. Common Placement Interview Questions

Q1. What is the difference between a substring and a subsequence?

A substring contains contiguous characters. A subsequence preserves character order but does not require contiguous positions.

Q2. How can you reverse a string without another array?

Use two pointers at the beginning and end, swapping characters until they meet.

Q3. How do you check whether two strings are anagrams efficiently?

Use a frequency array and compare character counts in O(n) time for a fixed character set.

Q4. How do you find the first non-repeating character?

Count frequencies in the first pass, then scan the original string in the second pass.

Q5. What is the sliding-window technique?

It maintains a range of elements and moves its boundaries to solve suitable contiguous-substring problems efficiently.

Q6. How can you check whether one string is a rotation of another?

Check whether the second string occurs in the first string concatenated with itself, after confirming equal lengths.

Q7. What is the difference between strcmp() and strstr()?

strcmp() compares two strings lexicographically. strstr() searches for the first occurrence of a substring.

Q8. Why should unsigned char be used as an index into a 256-element frequency array?

Because plain char may be signed. Converting to unsigned char ensures the index lies between 0 and 255.

Q9. Which algorithm is useful for finding a pattern in linear time?

KMP guarantees O(n + m) pattern-searching time.

Q10. How do you find the longest substring without repeating characters?

Use a sliding window and store the last-seen index of each character.

Q11. What is the difference between strlen() and sizeof() for a character array?

strlen() counts characters before the null terminator. sizeof() returns the total size in bytes of the array object when applied directly to an array.

Q12. What happens if a C string does not contain a null terminator?

String functions may read beyond the valid array bounds, resulting in undefined behavior.

Q13. What is the time complexity of reversing a string using two pointers?

O(n) time and O(1) auxiliary space.

Q14. How can you remove duplicate characters while preserving order?

Use a visited-character table and copy only the first occurrence of each character.

Q15. What is the difference between Rabin-Karp and KMP?

Rabin-Karp uses rolling hashes and may have O(nm) worst-case time. KMP uses an LPS array and guarantees O(n + m) time.
