2. Problem 1 – Subarray Sum Equals K

Problem: Given an integer array and a target K, count the number of contiguous subarrays whose sum equals K.

Example

Array = {1, 2, 3}
K = 3

Valid subarrays:

  • {1, 2}
  • {3}

Answer = 2.

Approach – Prefix Sum and Hashing

Let the prefix sum at position i be the sum of all elements from index 0 through i.

If:

current_prefix - previous_prefix = K

Then:

previous_prefix = current_prefix - K

Store the frequency of previously encountered prefix sums in a hash table. For each current prefix, add the frequency of current_prefix - K to the answer.

Complete C11 Program

#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <limits.h>
#include <stdbool.h>

#define TABLE_SIZE 101

typedef struct Node {
    long long key;
    long long frequency;
    struct Node *next;
} Node;

typedef struct {
    Node *bucket[TABLE_SIZE];
} HashTable;

unsigned int hash_ll(long long key)
{
    uint64_t value = (uint64_t)key;

    value ^= value >> 30;
    value *= UINT64_C(0xbf58476d1ce4e5b9);
    value ^= value >> 27;
    value *= UINT64_C(0x94d049bb133111eb);
    value ^= value >> 31;

    return (unsigned int)(value % TABLE_SIZE);
}

void init(HashTable *ht)
{
    for (int i = 0; i < TABLE_SIZE; i++) {
        ht->bucket[i] = NULL;
    }
}

long long get_frequency(HashTable *ht, long long key)
{
    unsigned int index = hash_ll(key);
    Node *current = ht->bucket[index];

    while (current != NULL) {
        if (current->key == key) {
            return current->frequency;
        }
        current = current->next;
    }

    return 0;
}

bool increment(HashTable *ht, long long key)
{
    unsigned int index = hash_ll(key);
    Node *current = ht->bucket[index];

    while (current != NULL) {
        if (current->key == key) {
            if (current->frequency == LLONG_MAX) {
                return false;
            }
            current->frequency++;
            return true;
        }
        current = current->next;
    }

    Node *newNode = malloc(sizeof(Node));

    if (newNode == NULL) {
        return false;
    }

    newNode->key = key;
    newNode->frequency = 1;
    newNode->next = ht->bucket[index];

    ht->bucket[index] = newNode;

    return true;
}

void destroy(HashTable *ht)
{
    for (int i = 0; i < TABLE_SIZE; i++) {
        Node *current = ht->bucket[i];

        while (current != NULL) {
            Node *temp = current;
            current = current->next;
            free(temp);
        }

        ht->bucket[i] = NULL;
    }
}

bool count_subarrays(
    const int arr[],
    int n,
    long long target,
    long long *answer)
{
    HashTable ht;
    init(&ht);

    long long prefix = 0;
    long long count = 0;

    /* Empty prefix has frequency one */
    if (!increment(&ht, 0)) {
        destroy(&ht);
        return false;
    }

    for (int i = 0; i < n; i++) {

        prefix += arr[i];

        long long required = prefix - target;

        long long frequency =
            get_frequency(&ht, required);

        if (count > LLONG_MAX - frequency) {
            destroy(&ht);
            return false;
        }

        count += frequency;

        if (!increment(&ht, prefix)) {
            destroy(&ht);
            return false;
        }
    }

    *answer = count;
    destroy(&ht);

    return true;
}

int main(void)
{
    int arr[] = {1, 2, 3};
    int n = sizeof(arr) / sizeof(arr[0]);

    long long answer;

    if (count_subarrays(arr, n, 3, &answer)) {
        printf("Subarrays with sum K: %lld\n", answer);
    } else {
        printf("Unable to calculate result.\n");
    }

    return 0;
}

Expected Output

Subarrays with sum K: 2

Complexity: O(n) expected time and O(n) extra space.

Interview tip: This method works even when the array contains negative numbers. A sliding-window solution based on expanding and shrinking the window is not generally valid for arrays with negative values.

3. Problem 2 – Longest Subarray with Sum Zero

Problem: Find the length of the longest contiguous subarray whose sum is zero.

Example

Array = {15, -2, 2, -8, 1, 7, 10, 23}

The subarray:

{-2, 2, -8, 1, 7}

has sum zero and length 5.

Approach

  • Calculate the running prefix sum.
  • Store the first index at which each prefix sum appears.
  • If the same prefix sum appears again, the elements between those indices have sum zero.
  • Update the maximum length.

Complete C11 Program – Hash Table

#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <stdbool.h>

#define TABLE_SIZE 101

typedef struct Node {
    long long key;
    int first_index;
    struct Node *next;
} Node;

typedef struct {
    Node *bucket[TABLE_SIZE];
} HashTable;

unsigned int hash_value(long long key)
{
    uint64_t x = (uint64_t)key;

    x ^= x >> 30;
    x *= UINT64_C(0xbf58476d1ce4e5b9);
    x ^= x >> 27;
    x *= UINT64_C(0x94d049bb133111eb);
    x ^= x >> 31;

    return (unsigned int)(x % TABLE_SIZE);
}

void init_table(HashTable *ht)
{
    for (int i = 0; i < TABLE_SIZE; i++) {
        ht->bucket[i] = NULL;
    }
}

bool find_index(
    HashTable *ht,
    long long key,
    int *index)
{
    unsigned int position = hash_value(key);
    Node *current = ht->bucket[position];

    while (current != NULL) {
        if (current->key == key) {
            *index = current->first_index;
            return true;
        }
        current = current->next;
    }

    return false;
}

bool insert_first(
    HashTable *ht,
    long long key,
    int index)
{
    unsigned int position = hash_value(key);

    Node *newNode = malloc(sizeof(Node));

    if (newNode == NULL) {
        return false;
    }

    newNode->key = key;
    newNode->first_index = index;
    newNode->next = ht->bucket[position];

    ht->bucket[position] = newNode;

    return true;
}

void destroy_table(HashTable *ht)
{
    for (int i = 0; i < TABLE_SIZE; i++) {
        Node *current = ht->bucket[i];

        while (current != NULL) {
            Node *temp = current;
            current = current->next;
            free(temp);
        }

        ht->bucket[i] = NULL;
    }
}

bool longest_zero_sum(
    const int arr[],
    int n,
    int *answer)
{
    HashTable ht;
    init_table(&ht);

    long long prefix = 0;
    int maximum = 0;

    /* Prefix sum zero exists before the array starts */
    if (!insert_first(&ht, 0, -1)) {
        destroy_table(&ht);
        return false;
    }

    for (int i = 0; i < n; i++) {
        prefix += arr[i];

        int previousIndex;

        if (find_index(&ht, prefix, &previousIndex)) {
            int length = i - previousIndex;

            if (length > maximum) {
                maximum = length;
            }
        } else {
            if (!insert_first(&ht, prefix, i)) {
                destroy_table(&ht);
                return false;
            }
        }
    }

    *answer = maximum;
    destroy_table(&ht);

    return true;
}

int main(void)
{
    int arr[] = {15, -2, 2, -8, 1, 7, 10, 23};
    int n = sizeof(arr) / sizeof(arr[0]);

    int answer;

    if (longest_zero_sum(arr, n, &answer)) {
        printf("Longest Zero Sum Subarray Length: %d\n",
               answer);
    } else {
        printf("Unable to calculate result.\n");
    }

    return 0;
}

Expected Output

Longest Zero Sum Subarray Length: 5

Complexity: O(n) expected time and O(n) extra space.

4. Problem 3 – Check Whether Two Strings Are Anagrams

Problem: Two strings are anagrams if they contain the same characters with the same frequencies, regardless of order.

Example

String 1 = "listen"
String 2 = "silent"

Output: Anagrams.

Complete C11 Program

#include <stdio.h>
#include <string.h>
#include <stdbool.h>

bool are_anagrams(const char a[], const char b[])
{
    int frequency[256] = {0};

    if (strlen(a) != strlen(b)) {
        return false;
    }

    for (int i = 0; a[i] != '\0'; i++) {
        unsigned char ch = (unsigned char)a[i];
        frequency[ch]++;
    }

    for (int i = 0; b[i] != '\0'; i++) {
        unsigned char ch = (unsigned char)b[i];

        frequency[ch]--;

        if (frequency[ch] < 0) {
            return false;
        }
    }

    return true;
}

int main(void)
{
    char str1[] = "listen";
    char str2[] = "silent";

    if (are_anagrams(str1, str2)) {
        printf("Strings are Anagrams.\n");
    } else {
        printf("Strings are Not Anagrams.\n");
    }

    return 0;
}

Expected Output

Strings are Anagrams.

Complexity: O(n) time and O(1) auxiliary space for the fixed 256-entry byte-frequency array.

This implementation compares bytes and is case-sensitive. It does not normalize uppercase/lowercase letters, ignore spaces, or process Unicode characters as complete code points.

5. Problem 4 – Find the Majority Element

Problem: Find an element that occurs more than n/2 times in an array, if such an element exists.

Example

Array = {2, 2, 1, 1, 2, 2, 2}

Majority element = 2.

Approach – Boyer-Moore Voting Algorithm

Although this problem can be solved using a hash table, the Boyer-Moore Voting Algorithm provides an O(n) time and O(1) extra-space solution.

Complete C11 Program

#include <stdio.h>
#include <stdbool.h>

bool majority_element(
    const int arr[],
    int n,
    int *answer)
{
    if (n <= 0) {
        return false;
    }

    int candidate = 0;
    int count = 0;

    /* Find a possible candidate */
    for (int i = 0; i < n; i++) {
        if (count == 0) {
            candidate = arr[i];
            count = 1;
        } else if (arr[i] == candidate) {
            count++;
        } else {
            count--;
        }
    }

    /* Verify the candidate */
    int frequency = 0;

    for (int i = 0; i < n; i++) {
        if (arr[i] == candidate) {
            frequency++;
        }
    }

    if (frequency > n / 2) {
        *answer = candidate;
        return true;
    }

    return false;
}

int main(void)
{
    int arr[] = {2, 2, 1, 1, 2, 2, 2};
    int n = sizeof(arr) / sizeof(arr[0]);

    int answer;

    if (majority_element(arr, n, &answer)) {
        printf("Majority Element: %d\n", answer);
    } else {
        printf("No Majority Element.\n");
    }

    return 0;
}

Expected Output

Majority Element: 2

Complexity: O(n) time and O(1) auxiliary space.

Interview point: The algorithm only identifies a candidate in its first pass. The second pass is essential because a majority element is not guaranteed to exist.

6. Problem 5 – Longest Subarray with Equal Number of 0s and 1s

Problem: Given a binary array containing only 0 and 1, find the length of the longest contiguous subarray containing equal numbers of both.

Example

Array = {0, 1, 0, 1, 1, 0, 0}

One longest valid subarray is:

{1, 0, 1, 1, 0, 0}

Answer = 6.

Approach

Convert each 0 into -1 and leave each 1 unchanged.

Now the problem becomes finding the longest zero-sum subarray. Reuse the prefix-sum hashing technique from Problem 2.

Complete C11 Program

#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <stdbool.h>

#define TABLE_SIZE 101

typedef struct Node {
    long long key;
    int first_index;
    struct Node *next;
} Node;

typedef struct {
    Node *bucket[TABLE_SIZE];
} HashTable;

unsigned int hash_key(long long key)
{
    uint64_t x = (uint64_t)key;

    x ^= x >> 30;
    x *= UINT64_C(0xbf58476d1ce4e5b9);
    x ^= x >> 27;
    x *= UINT64_C(0x94d049bb133111eb);
    x ^= x >> 31;

    return (unsigned int)(x % TABLE_SIZE);
}

void init(HashTable *ht)
{
    for (int i = 0; i < TABLE_SIZE; i++) {
        ht->bucket[i] = NULL;
    }
}

bool search(
    HashTable *ht,
    long long key,
    int *index)
{
    Node *current = ht->bucket[hash_key(key)];

    while (current != NULL) {
        if (current->key == key) {
            *index = current->first_index;
            return true;
        }
        current = current->next;
    }

    return false;
}

bool insert(
    HashTable *ht,
    long long key,
    int index)
{
    unsigned int position = hash_key(key);

    Node *node = malloc(sizeof(Node));

    if (node == NULL) {
        return false;
    }

    node->key = key;
    node->first_index = index;
    node->next = ht->bucket[position];

    ht->bucket[position] = node;

    return true;
}

void destroy(HashTable *ht)
{
    for (int i = 0; i < TABLE_SIZE; i++) {
        Node *current = ht->bucket[i];

        while (current != NULL) {
            Node *temp = current;
            current = current->next;
            free(temp);
        }

        ht->bucket[i] = NULL;
    }
}

bool longest_equal_zero_one(
    const int arr[],
    int n,
    int *answer)
{
    HashTable ht;
    init(&ht);

    long long prefix = 0;
    int maximum = 0;

    if (!insert(&ht, 0, -1)) {
        destroy(&ht);
        return false;
    }

    for (int i = 0; i < n; i++) {

        if (arr[i] != 0 && arr[i] != 1) {
            destroy(&ht);
            return false;
        }

        prefix += (arr[i] == 0) ? -1 : 1;

        int previous;

        if (search(&ht, prefix, &previous)) {
            int length = i - previous;

            if (length > maximum) {
                maximum = length;
            }
        } else {
            if (!insert(&ht, prefix, i)) {
                destroy(&ht);
                return false;
            }
        }
    }

    *answer = maximum;
    destroy(&ht);

    return true;
}

int main(void)
{
    int arr[] = {0, 1, 0, 1, 1, 0, 0};
    int n = sizeof(arr) / sizeof(arr[0]);

    int answer;

    if (longest_equal_zero_one(arr, n, &answer)) {
        printf("Longest Subarray Length: %d\n", answer);
    } else {
        printf("Invalid input or memory error.\n");
    }

    return 0;
}

Expected Output

Longest Subarray Length: 6

Complexity: O(n) expected time and O(n) extra space.

7. Problem 6 – Count Distinct Elements in Every Window

Problem: Given an array and a window size k, count the number of distinct elements in each contiguous window of size k.

Example

Array = {1, 2, 1, 3, 4, 2, 3}
k = 4
WindowDistinct Count
{1, 2, 1, 3}3
{2, 1, 3, 4}4
{1, 3, 4, 2}4
{3, 4, 2, 3}3

Expected output: 3 4 4 3

Approach – Sliding Window and Frequency Hashing

  1. Build the frequency table for the first window.
  2. Track how many keys currently have a nonzero frequency.
  3. When moving the window, decrease the frequency of the outgoing element.
  4. Remove it from the distinct count if its frequency becomes zero.
  5. Increase the frequency of the incoming element.
  6. Increase the distinct count if its frequency was previously zero.

Expected complexity: O(n) time and O(k) space with an appropriate frequency hash map.

8. Problem 7 – Group Anagrams

Problem: Given a list of strings, group strings that are anagrams of one another.

Example

Input:
eat, tea, tan, ate, nat, bat

Possible groups:
[eat, tea, ate]
[tan, nat]
[bat]

Approach

  • Build a frequency signature for each string.
  • Strings with identical signatures belong to the same group.
  • Use a hash table to map each signature to its group.

For lowercase English letters, the signature can be a sequence of 26 character counts.

Complexity: For m strings of maximum length L, frequency signatures take O(mL) time to construct, excluding the cost of hash-table collision comparisons and output storage.

9. Problem 8 – Find the Top K Frequent Elements

Problem: Given an array, return the k elements having the highest frequencies.

Example

Array = {1, 1, 1, 2, 2, 3}
k = 2

Answer: 1, 2

Approach

  1. Count frequencies using a hash map.
  2. Use a min heap of size k, or bucket sort based on frequency.
  3. Extract the k highest-frequency elements.

Complexity: Frequency counting is O(n) expected. A min heap of size k gives O(d log k) additional time, where d is the number of distinct values.

10. Problem 9 – Check Whether Two Arrays Are Equal

Problem: Determine whether two arrays contain exactly the same elements with the same frequencies, regardless of order.

Example

A = {1, 2, 2, 3}
B = {2, 3, 2, 1}

Output: Equal

Use a frequency hash table:

  1. Increase the frequency of every element in A.
  2. Decrease the frequency of every element in B.
  3. If the arrays have different lengths, return false immediately.
  4. If all frequencies become zero, the arrays are equal as multisets.

Expected complexity: O(n) time and O(n) extra space.

11. Problem 10 – Find the First Unique Character in a String

Problem: Find the first character that occurs exactly once.

Example

Input: "swiss"
Output: 'w'

Approach

  1. Count the frequency of every character.
  2. Scan the string from left to right.
  3. Return the first character with frequency one.

For a fixed byte alphabet, use a frequency array of size 256. The expected time is O(n), with O(1) auxiliary space.

13. Common Mistakes in Hashing Interviews

  1. Using a sliding window for arbitrary negative-number subarray-sum problems.
  2. Storing only the latest prefix-sum index when the problem requires the earliest one.
  3. Forgetting to initialize prefix sum zero with frequency one in subarray-sum counting.
  4. Overwriting the first occurrence of a prefix sum in longest-subarray problems.
  5. Ignoring duplicate elements when counting pairs.
  6. Assuming a majority element always exists without verifying the candidate.
  7. Using fixed-size hash tables without considering resizing and collision behavior.
  8. Ignoring integer overflow in prefix sums and pair calculations.

14. Placement Interview Questions and Answers

Q1. How can hashing solve the Subarray Sum Equals K problem?

Answer: Maintain a running prefix sum and store frequencies of previous prefix sums. For each prefix, add the frequency of prefix - K to the answer.

Q2. Why initialize the prefix-sum frequency of zero to one?

Answer: It accounts for subarrays beginning at index zero whose sum equals K.

Q3. How do you find the longest zero-sum subarray?

Answer: Store the first index at which each prefix sum occurs. When the same prefix sum appears again, calculate the distance between the two indices.

Q4. Why does converting zero to -1 help with equal 0s and 1s?

Answer: Equal counts of zeros and ones then produce a total sum of zero, so the problem becomes a longest zero-sum subarray problem.

Q5. What is the Boyer-Moore Voting Algorithm?

Answer: It finds a candidate majority element in O(n) time and O(1) space. A second pass verifies whether the candidate occurs more than n/2 times.

Q6. How can two arrays be checked for equality regardless of order?

Answer: Count the frequency of each value in the first array and subtract the frequencies of the second. All final frequencies must be zero.

Q7. What is the difference between a hash map and a hash set?

Answer: A hash map stores key-value pairs. A hash set stores unique keys and is primarily used for membership checking.

Q8. When should the first occurrence of a prefix sum be retained?

Answer: When finding the longest subarray for a given prefix-sum condition. Retaining the earliest index maximizes the possible subarray length.
