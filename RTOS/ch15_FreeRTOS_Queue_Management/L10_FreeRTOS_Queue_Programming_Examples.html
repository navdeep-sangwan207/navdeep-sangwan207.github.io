3. Example 1 – Basic Integer Queue Communication

Objective: Create a queue and transfer integer values between two tasks.

#include "FreeRTOS.h"
#include "task.h"
#include "queue.h"

QueueHandle_t dataQueue;

void SenderTask(void *pvParameters)
{
    int value = 10;

    for (;;)
    {
        if (xQueueSend(
                dataQueue,
                &value,
                pdMS_TO_TICKS(100)
            ) == pdPASS)
        {
            /* Item sent */
        }

        value += 10;

        vTaskDelay(pdMS_TO_TICKS(500));
    }
}

void ReceiverTask(void *pvParameters)
{
    int receivedValue;

    for (;;)
    {
        if (xQueueReceive(
                dataQueue,
                &receivedValue,
                portMAX_DELAY
            ) == pdPASS)
        {
            /* Process receivedValue */
        }
    }
}

int main(void)
{
    dataQueue = xQueueCreate(
        5,
        sizeof(int)
    );

    if (dataQueue == NULL)
    {
        for (;;)
        {
        }
    }

    xTaskCreate(
        SenderTask,
        "Sender",
        configMINIMAL_STACK_SIZE,
        NULL,
        2,
        NULL
    );

    xTaskCreate(
        ReceiverTask,
        "Receiver",
        configMINIMAL_STACK_SIZE,
        NULL,
        2,
        NULL
    );

    vTaskStartScheduler();

    for (;;)
    {
    }
}

Expected Result

The sender generates values such as 10, 20, 30 and 40. The receiver obtains them in FIFO order. The actual processing sequence depends on scheduling and queue availability.

4. Example 2 – Sending a Structure Through a Queue

Objective: Transfer multiple related sensor fields as one queue item.

#include "FreeRTOS.h"
#include "task.h"
#include "queue.h"

typedef struct
{
    int sensorID;
    int temperature;
    int humidity;
} SensorData_t;

QueueHandle_t sensorQueue;

{
    SensorData_t data;

    for (;;)
    {
        data.sensorID = 1;
        data.temperature = 28;
        data.humidity = 65;

        xQueueSend(
            sensorQueue,
            &data,
            pdMS_TO_TICKS(100)
        );

        vTaskDelay(pdMS_TO_TICKS(1000));

}

{
    SensorData_t receivedData;

    for (;;)
    {
        if (xQueueReceive(
                sensorQueue,
                &receivedData,
                portMAX_DELAY
            ) == pdPASS)
        {
            /* Process sensor fields */
        }
    }
}

int main(void)
{
    sensorQueue = xQueueCreate(
        5,
        sizeof(SensorData_t)
    );

    if (sensorQueue == NULL)
    {
        for (;;)
        {
        }
    }

    xTaskCreate(
        SensorTask,
        "Sensor",
        configMINIMAL_STACK_SIZE,
        NULL,
        2,
        NULL
    );

    xTaskCreate(
        ProcessingTask,
        "Process",
        configMINIMAL_STACK_SIZE,
        NULL,
        2,
        NULL
    );

    vTaskStartScheduler();

    for (;;)
    {
    }
}

Expected Result

The sensor task sends a structure containing sensor ID, temperature and humidity. The processing task receives a copy of the complete structure.

5. Example 3 – Multiple Producers and One Consumer

Objective: Transfer messages from two producer tasks to one consumer task.

#include "FreeRTOS.h"
#include "task.h"
#include "queue.h"

typedef struct
{
    int sourceID;
    int value;
} Message_t;

QueueHandle_t messageQueue;

void Producer1(void *pvParameters)
{
    Message_t message;

    message.sourceID = 1;

    for (;;)
    {
        message.value++;

        xQueueSend(
            messageQueue,
            &message,
            pdMS_TO_TICKS(100)
        );

        vTaskDelay(pdMS_TO_TICKS(300));
    }
}

void Producer2(void *pvParameters)
{
    Message_t message;

    message.sourceID = 2;

    for (;;)
    {
        message.value += 2;

        xQueueSend(
            messageQueue,
            &message,
            pdMS_TO_TICKS(100)
        );

        vTaskDelay(pdMS_TO_TICKS(500));
    }
}

void ConsumerTask(void *pvParameters)
{
    Message_t received;

    for (;;)
    {
        if (xQueueReceive(
                messageQueue,
                &received,
                portMAX_DELAY
            ) == pdPASS)
        {
            /* Process sourceID and value */
        }
    }
}

int main(void)
{
    messageQueue = xQueueCreate(
        10,
        sizeof(Message_t)
    );

    if (messageQueue == NULL)
    {
        for (;;)
        {
        }
    }

    xTaskCreate(
        Producer1,
        "Producer1",
        configMINIMAL_STACK_SIZE,
        NULL,
        2,
        NULL
    );

    xTaskCreate(
        Producer2,
        "Producer2",
        configMINIMAL_STACK_SIZE,
        NULL,
        2,
        NULL
    );

    xTaskCreate(
        ConsumerTask,
        "Consumer",
        configMINIMAL_STACK_SIZE,
        NULL,
        3,
        NULL
    );

    vTaskStartScheduler();

    for (;;)
    {
    }
}

Expected Result

Both producers send messages to the same queue. The consumer receives messages in the order they enter the queue. The sourceID field identifies the producer.

7. Example 5 – Queue Communication With Timeout

Objective: Handle queue-full and queue-empty conditions using finite timeouts.

void ProducerTask(void *pvParameters)
{
    int value = 1;
    BaseType_t status;

    for (;;)
    {
        status = xQueueSend(
            dataQueue,
            &value,
            pdMS_TO_TICKS(200)
        );

        if (status == pdPASS)
        {
            /* Send successful */
        }
        else
        {
            /* Timeout or queue remained full */
        }

        value++;

        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

void ConsumerTask(void *pvParameters)
{
    int receivedValue;
    BaseType_t status;

    for (;;)
    {
        status = xQueueReceive(
            dataQueue,
            &receivedValue,
            pdMS_TO_TICKS(500)
        );

        if (status == pdPASS)
        {
            /* Process receivedValue */
        }
        else
        {
            /* No item arrived before timeout */
        }
    }
}

Finite timeouts allow tasks to recover from missing data or a full queue rather than waiting indefinitely.

8. Example 6 – Queue Communication From an ISR

Objective: Transfer a small event record from interrupt context to a task.

#include "FreeRTOS.h"
#include "task.h"
#include "queue.h"

typedef struct
{
    unsigned int eventCode;
} EventMessage_t;

QueueHandle_t eventQueue;

void EventTask(void *pvParameters)
{
    EventMessage_t message;

    for (;;)
    {
        if (xQueueReceive(
                eventQueue,
                &message,
                portMAX_DELAY
            ) == pdPASS)
        {
            /* Process message.eventCode */
        }
    }
}

/* Illustrative ISR body */
void ApplicationISR(void)
{
    BaseType_t xHigherPriorityTaskWoken =
        pdFALSE;

    EventMessage_t message;

    message.eventCode = 100;

    if (xQueueSendFromISR(
            eventQueue,
            &message,
            &xHigherPriorityTaskWoken
        ) != pdPASS)
    {
        /* Record queue overflow */
    }

    portYIELD_FROM_ISR(
        xHigherPriorityTaskWoken
    );
}
The ISR declaration and yield macro are port-specific. Verify the actual FreeRTOS port's interrupt integration before using this example. The ISR must obey the port's interrupt-priority restrictions.

10. Example 8 – Queue Peek and Reset

Objective: Inspect the next queue item and clear queued data when required.

void QueueMaintenanceTask(void *pvParameters)
{
    int nextValue;
    BaseType_t status;

    for (;;)
    {
        status = xQueuePeek(
            dataQueue,
            &nextValue,
            0
        );

        if (status == pdPASS)
        {
            /* Inspect nextValue */
        }

        /*
         * Perform reset only as part of
         * a coordinated application action.
         */

        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

A reset operation should be performed only when the application has established that discarding pending items is safe. It should not be added to a periodic maintenance task without a defined requirement.

11. Example 9 – Queue Set With Two Queues

Objective: Wait for data from either of two queues using a queue set.

#include "FreeRTOS.h"
#include "task.h"
#include "queue.h"

QueueHandle_t queueA;
QueueHandle_t queueB;
QueueSetHandle_t queueSet;

void MonitorTask(void *pvParameters)
{
    QueueSetMemberHandle_t selected;
    int value;

    for (;;)
    {
        selected = xQueueSelectFromSet(
            queueSet,
            portMAX_DELAY
        );

        if (selected == queueA)
        {
            if (xQueueReceive(
                    queueA,
                    &value,
                    0
                ) == pdPASS)
            {
                /* Process queue A data */
            }
        }
        else if (selected == queueB)
        {
            if (xQueueReceive(
                    queueB,
                    &value,
                    0
                ) == pdPASS)
            {
                /* Process queue B data */
            }
        }
    }
}

void CreateQueueSet(void)
{
    queueA = xQueueCreate(5, sizeof(int));
    queueB = xQueueCreate(5, sizeof(int));

    if ((queueA == NULL) || (queueB == NULL))
    {
        return;
    }

    queueSet = xQueueCreateSet(10);

    if (queueSet == NULL)
    {
        return;
    }

    xQueueAddToSet(queueA, queueSet);
    xQueueAddToSet(queueB, queueSet);
}

The example demonstrates the selection pattern. A complete application should check the return values of queue-set membership operations and create the monitoring task after successful setup.

Queue-related problems may arise from incorrect queue configuration, insufficient capacity, incorrect item size or task synchronization errors.

13. Queue Design and Sizing

Queue capacity should be selected according to the maximum expected backlog rather than an arbitrary value.

Basic sizing idea:

Queue Capacity
    >= Maximum expected pending items

Approximate backlog
    = Arrival rate x Consumer delay

Additional capacity
    = Burst margin

For example, if a producer can generate 20 items per second and the consumer may be delayed for 0.5 seconds, approximately 10 items may accumulate during that delay. Additional capacity may be required for bursts and scheduling variation.

This is a preliminary sizing estimate, not a formal guarantee. Consider worst-case arrival bursts, service rates, blocking, and the application's data-loss policy.

14. Queue Programming Best Practices

  • Use meaningful queue and task names.
  • Check all queue creation and task creation results.
  • Use correct item sizes and data types.
  • Choose blocking times deliberately.
  • Define a policy for queue-full conditions.
  • Use structures for related fields when appropriate.
  • Define memory ownership when passing pointers.
  • Use FromISR APIs only in interrupt context.
  • Keep ISR processing short.
  • Do not use queue status snapshots as a substitute for checking operation results.
  • Coordinate queue reset and deletion with all users.
  • Consider static allocation in systems with strict memory requirements.

16. Practical Experiment

Objective: Compare queue behavior under different producer and consumer rates.

Procedure

  1. Create a queue with a small capacity, such as five items.
  2. Configure a producer task to send data periodically.
  3. Configure a consumer task with a longer processing delay.
  4. Observe whether the queue becomes full.
  5. Record failed send operations.
  6. Reduce the consumer delay and repeat.
  7. Increase the queue capacity and compare results.
  8. Use the debugger to inspect queue counts and task states.

Expected Result

When the producer generates data faster than the consumer processes it, the queue backlog increases. If the queue reaches capacity, a non-blocking send fails or a blocking send waits according to its timeout.

20. Chapter 15 – Final Summary

  • Queues provide buffered communication between tasks.
  • Queue items can contain integers, structures or pointer values.
  • Queue operations support blocking and non-blocking behavior.
  • Peek inspects data without removing it.
  • Reset clears queued items without deleting the queue.
  • Queue sets allow tasks to wait on multiple supported queue and semaphore members.
  • FromISR APIs support interrupt-context queue operations.
  • Queue capacity must account for bursts and consumer delays.
  • Reliable queue design requires correct memory ownership and error handling.
