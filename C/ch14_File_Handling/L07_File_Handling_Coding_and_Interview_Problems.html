3. Program 1: Create and Write Data into a File

This program creates a text file and writes multiple lines using fputs().

Complete C Program

#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    FILE *fp;

    fp = fopen("employee.txt", "w");

    if (fp == NULL) {
        perror("File creation failed");
        return EXIT_FAILURE;
    }

    if (fputs("Employee Management System\n", fp) == EOF ||
        fputs("Employee ID: 101\n", fp) == EOF ||
        fputs("Employee Name: Aman\n", fp) == EOF ||
        fputs("Department: Electronics\n", fp) == EOF) {

        perror("File writing failed");
        fclose(fp);
        return EXIT_FAILURE;
    }

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    printf("File created and data written successfully.\n");

    return EXIT_SUCCESS;
}

Expected Output

File created and data written successfully.

Contents of employee.txt

Employee Management System
Employee ID: 101
Employee Name: Aman
Department: Electronics

The w mode creates a new file or truncates an existing file before writing.

The append mode adds new content at the end of a file without removing its existing contents.

Append mode places output at the end of the stream. It is useful for log files, transaction records and audit trails.

5. Program 3: Count Characters, Words and Lines

This is a common placement-oriented file-processing problem.

Logic

  • Read the file character by character.
  • Increment the character count for every character.
  • Increment the line count whenever a newline is encountered.
  • Detect the beginning of a word using whitespace transitions.

Complete C Program

#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

int main(void)
{
    FILE *fp;
    int ch;

    long characters = 0;
    long words = 0;
    long lines = 0;

    int inside_word = 0;
    int last_char = EOF;

    fp = fopen("input.txt", "r");

    if (fp == NULL) {
        perror("File opening failed");
        return EXIT_FAILURE;
    }

    while ((ch = fgetc(fp)) != EOF) {

        characters++;
        last_char = ch;

        if (ch == '\n') {
            lines++;
        }

        if (isspace((unsigned char)ch)) {
            inside_word = 0;
        } else if (!inside_word) {
            words++;
            inside_word = 1;
        }
    }

    if (ferror(fp)) {
        perror("File reading failed");
        fclose(fp);
        return EXIT_FAILURE;
    }

    if (characters > 0 && last_char != '\n') {
        lines++;
    }

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    printf("Characters: %ld\n", characters);
    printf("Words: %ld\n", words);
    printf("Lines: %ld\n", lines);

    return EXIT_SUCCESS;
}

Example input.txt

C programming
File handling
Placement preparation

Expected Output

Characters: 45
Words: 6
Lines: 3

The counts shown assume the exact three lines above, with no extra newline after the last line.

This program counts bytes returned as characters from a text stream. For multibyte character encodings such as UTF-8, the count is not necessarily the number of user-perceived characters.

6. Program 4: Search for a Word in a File

This program searches for a word in each line of a text file using strstr().

Complete C Program

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(void)
{
    FILE *fp;
    char line[300];
    char target[] = "C";
    int found = 0;
    int line_number = 0;

    fp = fopen("input.txt", "r");

    if (fp == NULL) {
        perror("File opening failed");
        return EXIT_FAILURE;
    }

    while (fgets(line, sizeof line, fp) != NULL) {

        line_number++;

        if (strstr(line, target) != NULL) {
            printf("Match found at line %d: %s",
                   line_number, line);
            found = 1;
        }
    }

    if (ferror(fp)) {
        perror("File reading failed");
        fclose(fp);
        return EXIT_FAILURE;
    }

    if (!found) {
        printf("Word not found.\n");
    }

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

Example input.txt

C programming is useful.
Embedded C is used in firmware.
Linux supports C development.

Expected Output

Match found at line 1: C programming is useful.
Match found at line 2: Embedded C is used in firmware.

This search is case-sensitive and searches for a substring, not a complete word. For example, searching for "C" also matches the letter inside a larger word.

7. Program 5: Copy One Text File into Another

This program copies the contents of one text file into another using fgets() and fputs().

Complete C Program

#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    FILE *source;
    FILE *destination;

    char buffer[512];
    int status = EXIT_SUCCESS;

    source = fopen("source.txt", "r");

    if (source == NULL) {
        perror("Source file opening failed");
        return EXIT_FAILURE;
    }

    destination = fopen("destination.txt", "w");

    if (destination == NULL) {
        perror("Destination file opening failed");
        fclose(source);
        return EXIT_FAILURE;
    }

    while (fgets(buffer, sizeof buffer, source) != NULL) {

        if (fputs(buffer, destination) == EOF) {
            perror("File copying failed");
            status = EXIT_FAILURE;
            break;
        }
    }

    if (ferror(source)) {
        perror("Source reading failed");
        status = EXIT_FAILURE;
    }

    if (fclose(source) == EOF) {
        perror("Source closing failed");
        status = EXIT_FAILURE;
    }

    if (fclose(destination) == EOF) {
        perror("Destination closing failed");
        status = EXIT_FAILURE;
    }

    if (status == EXIT_SUCCESS) {
        printf("File copied successfully.\n");
    }

    return status;
}

Expected Output

File copied successfully.
This example is for text files. For arbitrary binary files such as images, use fread() and fwrite() in binary mode.

8. Program 6: Merge Two Text Files

Merging means combining the contents of two input files into one output file.

Complete C Program

#include <stdio.h>
#include <stdlib.h>

int copy_file(FILE *source, FILE *destination)
{
    int ch;

    while ((ch = fgetc(source)) != EOF) {
        if (fputc(ch, destination) == EOF) {
            return 0;
        }
    }

    if (ferror(source)) {
        return 0;
    }

    return 1;
}

int main(void)
{
    FILE *fp1;
    FILE *fp2;
    FILE *output;

    int status = EXIT_SUCCESS;

    fp1 = fopen("first.txt", "r");

    if (fp1 == NULL) {
        perror("First file opening failed");
        return EXIT_FAILURE;
    }

    fp2 = fopen("second.txt", "r");

    if (fp2 == NULL) {
        perror("Second file opening failed");
        fclose(fp1);
        return EXIT_FAILURE;
    }

    output = fopen("merged.txt", "w");

    if (output == NULL) {
        perror("Output file opening failed");
        fclose(fp1);
        fclose(fp2);
        return EXIT_FAILURE;
    }

    if (!copy_file(fp1, output) ||
        !copy_file(fp2, output)) {
        fprintf(stderr, "Merging failed.\n");
        status = EXIT_FAILURE;
    }

    if (fclose(fp1) == EOF) {
        perror("First file closing failed");
        status = EXIT_FAILURE;
    }

    if (fclose(fp2) == EOF) {
        perror("Second file closing failed");
        status = EXIT_FAILURE;
    }

    if (fclose(output) == EOF) {
        perror("Output file closing failed");
        status = EXIT_FAILURE;
    }

    if (status == EXIT_SUCCESS) {
        printf("Files merged successfully.\n");
    }

    return status;
}

Example Input

first.txt:

First file content.
C programming.

second.txt:

Second file content.
File handling.

Expected Output

Files merged successfully.

merged.txt contains the contents of first.txt followed by second.txt.

9. Program 7: Store and Display Student Records

This program demonstrates a simple file-based student record system using structures and formatted text files.

Complete C Program

#include <stdio.h>
#include <stdlib.h>

#define COUNT 3

struct Student {
    int roll;
    char name[40];
    float marks;
};

int main(void)
{
    struct Student s;
    FILE *fp;

    fp = fopen("students.txt", "w");

    if (fp == NULL) {
        perror("File creation failed");
        return EXIT_FAILURE;
    }

    for (int i = 0; i < COUNT; i++) {

        printf("Enter roll number, name and marks: ");

        if (scanf("%d %39s %f",
                  &s.roll, s.name, &s.marks) != 3) {
            fprintf(stderr, "Invalid input.\n");
            fclose(fp);
            return EXIT_FAILURE;
        }

        if (fprintf(fp, "%d %s %.2f\n",
                    s.roll, s.name, s.marks) < 0) {
            perror("Writing record failed");
            fclose(fp);
            return EXIT_FAILURE;
        }
    }

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    fp = fopen("students.txt", "r");

    if (fp == NULL) {
        perror("File opening failed");
        return EXIT_FAILURE;
    }

    printf("\nStored Student Records:\n");

    while (fscanf(fp, "%d %39s %f",
                  &s.roll, s.name, &s.marks) == 3) {

        printf("%d %s %.2f\n",
               s.roll, s.name, s.marks);
    }

    if (ferror(fp)) {
        perror("Reading records failed");
        fclose(fp);
        return EXIT_FAILURE;
    }

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

Example Input

101 Aman 88.5
102 Riya 91.0
103 Karan 85.5

Expected Output

Stored Student Records:
101 Aman 88.50
102 Riya 91.00
103 Karan 85.50

This demonstration uses names without spaces because the %s conversion stops at whitespace. A production application should validate input more thoroughly and use a clearly defined record format.

10. Program 8: Search for a Student Record

This program searches a text file for a student using the roll number.

Complete C Program

#include <stdio.h>
#include <stdlib.h>

struct Student {
    int roll;
    char name[40];
    float marks;
};

int main(void)
{
    FILE *fp;
    struct Student s;

    int search_roll;
    int found = 0;

    fp = fopen("students.txt", "r");

    if (fp == NULL) {
        perror("File opening failed");
        return EXIT_FAILURE;
    }

    printf("Enter roll number to search: ");

    if (scanf("%d", &search_roll) != 1) {
        fprintf(stderr, "Invalid roll number.\n");
        fclose(fp);
        return EXIT_FAILURE;
    }

    while (fscanf(fp, "%d %39s %f",
                  &s.roll, s.name, &s.marks) == 3) {

        if (s.roll == search_roll) {

            printf("\nStudent Found\n");
            printf("Roll: %d\n", s.roll);
            printf("Name: %s\n", s.name);
            printf("Marks: %.2f\n", s.marks);

            found = 1;
            break;
        }
    }

    if (ferror(fp)) {
        perror("File reading failed");
        fclose(fp);
        return EXIT_FAILURE;
    }

    if (!found) {
        printf("Student record not found.\n");
    }

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

Example Input

Expected Output

11. Program 9: Count the Frequency of a Character in a File

This program counts how many times a particular character occurs in a text file.

Complete C Program

#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    FILE *fp;

    int ch;
    int target;
    long count = 0;

    fp = fopen("input.txt", "r");

    if (fp == NULL) {
        perror("File opening failed");
        return EXIT_FAILURE;
    }

    printf("Enter character to search: ");

    target = getchar();

    if (target == EOF) {
        fprintf(stderr, "No character supplied.\n");
        fclose(fp);
        return EXIT_FAILURE;
    }

    while ((ch = fgetc(fp)) != EOF) {
        if (ch == target) {
            count++;
        }
    }

    if (ferror(fp)) {
        perror("File reading failed");
        fclose(fp);
        return EXIT_FAILURE;
    }

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    printf("Frequency of '%c' = %ld\n", target, count);

    return EXIT_SUCCESS;
}

Example Input

Enter character to search: a

Expected Output

Frequency of 'a' = 5

The count depends on the contents of input.txt.

12. Program 10: Random Access and Update a Binary Record

This program creates employee records, searches for a particular ID, and updates the salary of the matching record.

Complete C Program

#include <stdio.h>
#include <stdlib.h>

#define COUNT 3

struct Employee {
    int id;
    char name[30];
    float salary;
};

int main(void)
{
    struct Employee employees[COUNT] = {
        {101, "Aman", 25000.0f},
        {102, "Riya", 30000.0f},
        {103, "Karan", 28000.0f}
    };

    struct Employee employee;

    FILE *fp;
    int search_id = 102;
    int found = 0;

    fp = fopen("employees.bin", "wb");

    if (fp == NULL) {
        perror("File creation failed");
        return EXIT_FAILURE;
    }

    if (fwrite(employees, sizeof employees[0],
               COUNT, fp) != COUNT) {
        fprintf(stderr, "Writing records failed.\n");
        fclose(fp);
        return EXIT_FAILURE;
    }

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    fp = fopen("employees.bin", "rb+");

    if (fp == NULL) {
        perror("File opening failed");
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < COUNT; i++) {

        if (fseek(fp,
                  (long int)(i * sizeof employee),
                  SEEK_SET) != 0) {
            perror("Positioning failed");
            fclose(fp);
            return EXIT_FAILURE;
        }

        if (fread(&employee, sizeof employee, 1, fp) != 1) {
            fprintf(stderr, "Record reading failed.\n");
            fclose(fp);
            return EXIT_FAILURE;
        }

        if (employee.id == search_id) {

            employee.salary = 45000.0f;

            if (fseek(fp,
                      (long int)(i * sizeof employee),
                      SEEK_SET) != 0) {
                perror("Positioning failed");
                fclose(fp);
                return EXIT_FAILURE;
            }

            if (fwrite(&employee, sizeof employee, 1, fp) != 1) {
                fprintf(stderr, "Record updating failed.\n");
                fclose(fp);
                return EXIT_FAILURE;
            }

            found = 1;
            break;
        }
    }

    if (!found) {
        printf("Employee not found.\n");
    } else {
        printf("Employee salary updated successfully.\n");
    }

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    return found ? EXIT_SUCCESS : EXIT_FAILURE;
}

Expected Output

Employee salary updated successfully.

The example uses raw structure storage for a controlled demonstration. It assumes compatible structure layout when writing and reading the file.

13. Program 11: Count the Number of Records in a Binary File

For a binary file containing fixed-size records, the file size can be divided by the size of one record, provided the file is known to contain complete records in the same representation.

Complete C Program

#include <stdio.h>
#include <stdlib.h>

struct Student {
    int roll;
    char name[30];
    float marks;
};

int main(void)
{
    FILE *fp;
    long int file_size;
    size_t record_size;
    long int record_count;

    fp = fopen("students.bin", "rb");

    if (fp == NULL) {
        perror("File opening failed");
        return EXIT_FAILURE;
    }

    if (fseek(fp, 0L, SEEK_END) != 0) {
        perror("File positioning failed");
        fclose(fp);
        return EXIT_FAILURE;
    }

    file_size = ftell(fp);

    if (file_size == -1L) {
        perror("File size retrieval failed");
        fclose(fp);
        return EXIT_FAILURE;
    }

    record_size = sizeof(struct Student);

    if (file_size % (long int)record_size != 0) {
        fprintf(stderr, "File contains an incomplete record.\n");
        fclose(fp);
        return EXIT_FAILURE;
    }

    record_count = file_size / (long int)record_size;

    printf("File size: %ld bytes\n", file_size);
    printf("Record size: %zu bytes\n", record_size);
    printf("Number of records: %ld\n", record_count);

    if (fclose(fp) == EOF) {
        perror("File closing failed");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

The program assumes a seekable file and a file created using the same record representation. For large files, portable C's long-based positioning may be insufficient.

14. Program 12: Compare Two Text Files

This program compares two text files character by character and reports whether their contents are identical.

Complete C Program

#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    FILE *fp1;
    FILE *fp2;

    int ch1;
    int ch2;
    int identical = 1;

    fp1 = fopen("file1.txt", "r");

    if (fp1 == NULL) {
        perror("First file opening failed");
        return EXIT_FAILURE;
    }

    fp2 = fopen("file2.txt", "r");

    if (fp2 == NULL) {
        perror("Second file opening failed");
        fclose(fp1);
        return EXIT_FAILURE;
    }

    while (1) {

        ch1 = fgetc(fp1);
        ch2 = fgetc(fp2);

        if (ch1 == EOF || ch2 == EOF) {
            break;
        }

        if (ch1 != ch2) {
            identical = 0;
            break;
        }
    }

    if (ferror(fp1) || ferror(fp2)) {
        perror("File reading failed");
        identical = 0;
    } else if (ch1 != ch2) {
        identical = 0;
    }

    if (fclose(fp1) == EOF) {
        perror("First file closing failed");
        identical = 0;
    }

    if (fclose(fp2) == EOF) {
        perror("Second file closing failed");
        identical = 0;
    }

    if (identical) {
        printf("Files are identical.\n");
    } else {
        printf("Files are different or could not be compared.\n");
    }

    return identical ? EXIT_SUCCESS : EXIT_FAILURE;
}

Expected Output

Files are identical.

The result assumes both files contain identical text. If the file contents differ, the program reports that they are different.

15. Common File Handling Interview Questions

Q1. What is the difference between w and a modes?

Answer: w creates or truncates a file, while a opens or creates a file and places output at the end.

Q2. What is the difference between r and rb?

Answer: Both open an existing file for reading. rb requests binary mode, which avoids text-stream translations where the implementation distinguishes text and binary streams.

Q3. Why should fgetc() return value be stored in int?

Answer: To represent all possible character values and the special EOF value.

Q4. What is the difference between fgets() and fscanf()?

Answer: fgets() reads a bounded character sequence, usually a line, while fscanf() performs formatted input according to a format string.

Q5. What is the difference between fprintf() and fwrite()?

Answer: fprintf() converts data into formatted text, while fwrite() writes blocks of bytes from memory.

Q6. Can fwrite() store a complete structure?

Answer: Yes. It can write the structure's object representation, but this does not guarantee portability across different implementations.

Q7. What is random file access?

Answer: It is the ability to reposition a stream and access a selected location without sequentially processing all preceding data.

Q8. What is the purpose of fclose()?

Answer: It closes the stream and performs the required flushing of buffered output.

Q9. What is the difference between EOF and NULL?

Answer: EOF is a special negative integer constant used by input functions. NULL is a null pointer constant used to represent an invalid or absent pointer value.

Q10. Why should file operations be checked?

Answer: File operations can fail because of missing files, insufficient permissions, storage problems, read errors and write failures.

Problem 2

Question: What will be stored in the file?

Answer: 10 20

Problem 3

int ch;

while ((ch = fgetc(fp)) != EOF) {
    count++;
}

Question: What does count represent?

Answer: The number of characters returned by the text stream's input operations.

Problem 4

fseek(fp, 0L, SEEK_END);
long int size = ftell(fp);

Question: What does size represent for a seekable binary file?

Answer: The file position at the end, commonly interpreted as the file size in bytes for an ordinary binary file.

Problem 5

while (!feof(fp)) {
    ch = fgetc(fp);
}

Question: Identify the problem.

Answer: The loop tests the EOF indicator before the next read. The indicator is set only after an input operation encounters the end of the stream, so the final read result must be checked directly.
