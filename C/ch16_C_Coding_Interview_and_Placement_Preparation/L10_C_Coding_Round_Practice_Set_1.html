2. Number-Based Programming

Problem 1: Check whether a number is prime

Difficulty: Easy

Problem statement: Write a program to determine whether a given integer is prime.

Input: One integer n.

Constraint: 0 ≤ n ≤ INT_MAX.

Sample input
29
Sample output
Prime
Show C11 solution 
#include <stdio.h>

int main(void)
{
    int n;
    int is_prime = 1;

    if (scanf("%d", &n) != 1)
        return 1;

    if (n < 2)
        is_prime = 0;

    for (int i = 2; is_prime && i <= n / i; i++)
    {
        if (n % i == 0)
            is_prime = 0;
    }

    if (is_prime)
        printf("Prime\n");
    else
        printf("Not prime\n");

    return 0;
}

Complexity: O(√n) time and O(1) auxiliary space.

Important: The condition i ≤ n / i avoids the possible overflow of i * i.

Problem 2: Find the GCD of two numbers

Difficulty: Easy

Problem statement: Find the greatest common divisor of two non-negative integers using the Euclidean algorithm.

Sample input
48 18
Sample output
GCD = 6
Show C11 solution 
#include <stdio.h>

int main(void)
{
    int a, b;

    if (scanf("%d %d", &a, &b) != 2)
        return 1;

    if (a < 0 || b < 0)
    {
        printf("Invalid input\n");
        return 1;
    }

    int x = a;
    int y = b;

    while (y != 0)
    {
        int remainder = x % y;
        x = y;
        y = remainder;
    }

    printf("GCD = %d\n", x);

    return 0;
}

Complexity: O(log(min(a,b))) time and O(1) auxiliary space.

By convention, this implementation returns 0 for GCD(0, 0).

Problem 3: Check whether a number is an Armstrong number

Difficulty: Easy to Medium

Problem statement: Determine whether a non-negative integer is equal to the sum of its digits, each raised to the number of digits.

For example, 153 = 1³ + 5³ + 3³.

Sample input
153
Sample output
Armstrong number
Show C11 solution 
#include <stdio.h>

int main(void)
{
    unsigned int n;

    if (scanf("%u", &n) != 1)
        return 1;

    unsigned int temp = n;
    unsigned int digits = 0;

    do
    {
        digits++;
        temp /= 10U;
    } while (temp != 0U);

    temp = n;

    unsigned long long sum = 0;

    do
    {
        unsigned int digit = temp % 10U;
        unsigned long long power = 1;

        for (unsigned int i = 0; i < digits; i++)
            power *= digit;

        sum += power;
        temp /= 10U;

    } while (temp != 0U);

    if (sum == n)
        printf("Armstrong number\n");
    else
        printf("Not an Armstrong number\n");

    return 0;
}

Complexity: O(d²) time in this straightforward implementation, where d is the number of digits; O(1) auxiliary space.

Note: The program uses unsigned long long for the sum. For unrestricted input sizes, intermediate arithmetic would require additional overflow handling.

3. Array Programming

Problem 4: Find the largest and smallest elements

Difficulty: Easy

Problem statement: Read an array and find its minimum and maximum values.

Sample input
6
12 5 38 2 19 27
Sample output
Minimum = 2
Maximum = 38
Show C11 solution 
#include <stdio.h>

int main(void)
{
    int n;

    if (scanf("%d", &n) != 1 || n <= 0 || n > 1000)
        return 1;

    int arr[1000];

    for (int i = 0; i < n; i++)
    {
        if (scanf("%d", &arr[i]) != 1)
            return 1;
    }

    int minimum = arr[0];
    int maximum = arr[0];

    for (int i = 1; i < n; i++)
    {
        if (arr[i] < minimum)
            minimum = arr[i];

        if (arr[i] > maximum)
            maximum = arr[i];
    }

    printf("Minimum = %d\n", minimum);
    printf("Maximum = %d\n", maximum);

    return 0;
}

Complexity: O(n) time and O(n) space for the input array. The search itself uses O(1) auxiliary space.

Problem 5: Move all zeros to the end

Difficulty: Medium

Problem statement: Move every zero to the end of an array while preserving the relative order of non-zero elements.

Sample input
7
0 4 0 2 8 0 5
Sample output
4 2 8 5 0 0 0
Show C11 solution 
#include <stdio.h>

int main(void)
{
    int n;

    if (scanf("%d", &n) != 1 || n < 0 || n > 1000)
        return 1;

    int arr[1000];

    for (int i = 0; i < n; i++)
    {
        if (scanf("%d", &arr[i]) != 1)
            return 1;
    }

    int position = 0;

    for (int i = 0; i < n; i++)
    {
        if (arr[i] != 0)
            arr[position++] = arr[i];
    }

    while (position < n)
        arr[position++] = 0;

    for (int i = 0; i < n; i++)
        printf("%d ", arr[i]);

    printf("\n");
    return 0;
}

Complexity: O(n) time and O(1) auxiliary space.

Problem 6: Find the missing number

Difficulty: Medium

Problem statement: An array contains distinct integers from 1 through n, with exactly one number missing. Find the missing number.

Input format: First enter n, followed by n−1 array elements.

Sample input
5
1 2 4 5
Sample output
Missing number = 3
Show C11 solution 
#include <stdio.h>

int main(void)
{
    unsigned int n;

    if (scanf("%u", &n) != 1 || n == 0U || n > 1000000U)
        return 1;

    unsigned int missing = 0U;

    for (unsigned int i = 1U; i <= n; i++)
        missing ^= i;

    for (unsigned int i = 0U; i < n - 1U; i++)
    {
        unsigned int value;

        if (scanf("%u", &value) != 1 || value < 1U || value > n)
            return 1;

        missing ^= value;
    }

    printf("Missing number = %u\n", missing);

    return 0;
}

Complexity: O(n) time and O(1) auxiliary space.

Assumption: Input values are distinct and exactly one value in the range is missing. XOR cancels matching values.

4. String Programming

Problem 7: Count vowels, consonants, digits and spaces

Difficulty: Easy

Problem statement: Read a line and count vowels, consonants, digits and spaces.

Sample input
Hello C 123
Sample output
Vowels = 2
Consonants = 4
Digits = 3
Spaces = 2
Show C11 solution 
#include <stdio.h>
#include <ctype.h>

int main(void)
{
    char str[500];

    if (fgets(str, sizeof(str), stdin) == NULL)
        return 1;

    int vowels = 0;
    int consonants = 0;
    int digits = 0;
    int spaces = 0;

    for (int i = 0; str[i] != '\0'; i++)
    {
        unsigned char ch = (unsigned char)str[i];

        if (ch == '\n')
            continue;

        if (isalpha(ch))
        {
            ch = (unsigned char)tolower(ch);

            if (ch == 'a' || ch == 'e' || ch == 'i' ||
                ch == 'o' || ch == 'u')
                vowels++;
            else
                consonants++;
        }
        else if (isdigit(ch))
        {
            digits++;
        }
        else if (ch == ' ')
        {
            spaces++;
        }
    }

    printf("Vowels = %d\n", vowels);
    printf("Consonants = %d\n", consonants);
    printf("Digits = %d\n", digits);
    printf("Spaces = %d\n", spaces);

    return 0;
}

Complexity: O(n) time and O(1) auxiliary space, excluding the fixed input buffer.

Problem 8: Remove duplicate characters from a string

Difficulty: Medium

Problem statement: Remove repeated bytes while preserving the first occurrence of each character.

Sample input
programming
Sample output
progamin
Show C11 solution 
#include <stdio.h>

int main(void)
{
    char str[500];
    unsigned char seen[256] = {0};

    if (fgets(str, sizeof(str), stdin) == NULL)
        return 1;

    size_t read_index = 0;
    size_t write_index = 0;

    while (str[read_index] != '\0')
    {
        unsigned char ch = (unsigned char)str[read_index];

        if (ch == '\n')
            break;

        if (!seen[ch])
        {
            seen[ch] = 1;
            str[write_index++] = (char)ch;
        }

        read_index++;
    }

    str[write_index] = '\0';

    printf("%s\n", str);

    return 0;
}

Complexity: O(n) time and O(1) auxiliary space because the lookup table has a fixed 256 entries.

This solution treats the input as bytes, not Unicode characters.

Problem 9: Find the first occurrence of a substring

Difficulty: Medium

Problem statement: Find the first occurrence of a pattern in a text without using strstr().

Sample input
embedded systems
systems
Sample output
Found at index 9
Show C11 solution 
#include <stdio.h>
#include <string.h>

int main(void)
{
    char text[500];
    char pattern[200];

    if (fgets(text, sizeof(text), stdin) == NULL ||
        fgets(pattern, sizeof(pattern), stdin) == NULL)
        return 1;

    text[strcspn(text, "\n")] = '\0';
    pattern[strcspn(pattern, "\n")] = '\0';

    size_t text_length = strlen(text);
    size_t pattern_length = strlen(pattern);

    if (pattern_length == 0)
    {
        printf("Found at index 0\n");
        return 0;
    }

    int found = 0;

    if (pattern_length <= text_length)
    {
        for (size_t i = 0; i <= text_length - pattern_length; i++)
        {
            size_t j = 0;

            while (j < pattern_length &&
                   text[i + j] == pattern[j])
                j++;

            if (j == pattern_length)
            {
                printf("Found at index %zu\n", i);
                found = 1;
                break;
            }
        }
    }

    if (!found)
        printf("Not found\n");

    return 0;
}

Complexity: O(n × m) worst-case time, where n is the text length and m is the pattern length; O(1) auxiliary space.

5. Pointer and Memory Programming

Problem 10: Swap two integers using pointers

Difficulty: Easy

Sample input
10 20
Sample output
After swapping: 20 10
Show C11 solution 
#include <stdio.h>

void swap(int *a, int *b)
{
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void)
{
    int x, y;

    if (scanf("%d %d", &x, &y) != 2)
        return 1;

    swap(&x, &y);

    printf("After swapping: %d %d\n", x, y);

    return 0;
}

Complexity: O(1) time and O(1) auxiliary space.

The function receives valid pointers to two integer objects. It modifies the original objects through indirection.

Problem 11: Find the sum of an array using pointer arithmetic

Difficulty: Easy to Medium

Sample input
5
2 4 6 8 10
Sample output
Sum = 30
Show C11 solution 
#include <stdio.h>

int main(void)
{
    int n;

    if (scanf("%d", &n) != 1 || n <= 0 || n > 1000)
        return 1;

    int arr[1000];
    int *ptr = arr;

    for (int i = 0; i < n; i++)
    {
        if (scanf("%d", ptr + i) != 1)
            return 1;
    }

    int sum = 0;

    for (int i = 0; i < n; i++)
        sum += *(ptr + i);

    printf("Sum = %d\n", sum);

    return 0;
}

Complexity: O(n) time and O(n) space for the array; O(1) auxiliary space.

All pointer arithmetic stays within the array object.

6. Bit Manipulation

Problem 12: Check whether a number is a power of two

Difficulty: Easy

Sample input
64
Sample output
Power of two
Show C11 solution 
#include <stdio.h>

int main(void)
{
    unsigned int n;

    if (scanf("%u", &n) != 1)
        return 1;

    if (n != 0U && (n & (n - 1U)) == 0U)
        printf("Power of two\n");
    else
        printf("Not a power of two\n");

    return 0;
}

Complexity: O(1) time and O(1) auxiliary space.

A positive power of two has exactly one set bit in its binary representation.

Problem 13: Swap two unsigned integers using XOR

Difficulty: Easy

Sample input
12 25
Sample output
25 12
Show C11 solution 
#include <stdio.h>

int main(void)
{
    unsigned int a, b;

    if (scanf("%u %u", &a, &b) != 2)
        return 1;

    if (&a != &b)
    {
        a ^= b;
        b ^= a;
        a ^= b;
    }

    printf("%u %u\n", a, b);

    return 0;
}

Complexity: O(1) time and O(1) auxiliary space.

In this example a and b are separate local objects, so the address check is always true. The check is important when implementing XOR swap through pointers because passing the same object twice would otherwise zero it.

7. Basic Data Structures

Problem 14: Implement a stack using an array

Difficulty: Medium

Problem statement: Implement push, pop and display operations using a fixed-size integer array.

#include <stdio.h>

#define MAX 5

int stack[MAX];
int top = -1;

int push(int value)
{
    if (top == MAX - 1)
        return 0;

    stack[++top] = value;
    return 1;
}

int pop(int *value)
{
    if (top == -1)
        return 0;

    *value = stack[top--];
    return 1;
}

void display(void)
{
    if (top == -1)
    {
        printf("Stack is empty\n");
        return;
    }

    for (int i = top; i >= 0; i--)
        printf("%d ", stack[i]);

    printf("\n");
}

int main(void)
{
    int value;

    push(10);
    push(20);
    push(30);

    display();

    if (pop(&value))
        printf("Popped: %d\n", value);

    display();

    return 0;
}
Expected output
30 20 10
Popped: 30
20 10

Complexity: Push O(1), pop O(1), display O(n). The stack uses O(MAX) storage.

Problem 15: Reverse a singly linked list

Difficulty: Medium

Problem statement: Reverse the direction of links in a singly linked list.

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

struct Node *reverse(struct Node *head)
{
    struct Node *previous = NULL;
    struct Node *current = head;

    while (current != NULL)
    {
        struct Node *next_node = current->next;

        current->next = previous;
        previous = current;
        current = next_node;
    }

    return previous;
}

void display(const struct Node *head)
{
    while (head != NULL)
    {
        printf("%d ", head->data);
        head = head->next;
    }

    printf("\n");
}

void free_list(struct Node *head)
{
    while (head != NULL)
    {
        struct Node *next_node = head->next;
        free(head);
        head = next_node;
    }
}

int main(void)
{
    int values[] = {10, 20, 30, 40};
    struct Node *head = NULL;
    struct Node *tail = NULL;

    for (size_t i = 0; i < sizeof(values) / sizeof(values[0]); i++)
    {
        struct Node *node = malloc(sizeof(*node));

        if (node == NULL)
        {
            free_list(head);
            return 1;
        }

        node->data = values[i];
        node->next = NULL;

        if (head == NULL)
            head = node;
        else
            tail->next = node;

        tail = node;
    }

    printf("Original list: ");
    display(head);

    head = reverse(head);

    printf("Reversed list: ");
    display(head);

    free_list(head);

    return 0;
}
Expected output
Original list: 10 20 30 40
Reversed list: 40 30 20 10

Complexity: O(n) time and O(1) auxiliary space for reversal. The linked list itself requires O(n) storage.

9. Compilation and Debugging Practice

Use GCC or a compatible C11 compiler to compile your solutions.

gcc -std=c11 -Wall -Wextra -Wpedantic -g program.c -o program

Run the program from the terminal:

./program

For additional runtime checks, where supported by your compiler:

gcc -std=c11 -Wall -Wextra -Wpedantic -g \
-fsanitize=address,undefined program.c -o program
Sanitizers are compiler-toolchain features, not part of the C11 language standard. A program that completes successfully under sanitizers is not automatically proven correct.

12. Final Revision Checklist

  • Can you write each solution without copying?
  • Can you explain the algorithm before coding?
  • Can you identify the time and space complexity?
  • Can you handle empty, minimum and maximum input cases?
  • Can you explain pointer and memory operations?
  • Can you compile with warnings and fix them?
  • Can you debug an incorrect result systematically?
  • Can you explain how the same concepts apply to embedded C?
