2. Arrays and Searching

Problem 1: Find a pair with a given sum in a sorted array

Difficulty: Medium

Concepts: Two-pointer technique, sorted arrays, overflow-aware arithmetic.

Problem statement: Given a sorted array and a target value, determine whether two distinct elements add up to the target. Print their values if a pair exists.

Sample input 
6
1 3 4 6 8 10
14
Sample output 
Pair found: 4 10
Show C11 solution 
#include <stdio.h>

int main(void)
{
    int n;
    int arr[1000];
    long long target;

    if (scanf("%d", &n) != 1 || n < 0 || n > 1000)
        return 1;

    for (int i = 0; i < n; i++)
    {
        if (scanf("%d", &arr[i]) != 1)
            return 1;

        if (i > 0 && arr[i] < arr[i - 1])
        {
            printf("Array must be sorted\n");
            return 1;
        }
    }

    if (scanf("%lld", &target) != 1)
        return 1;

    int left = 0;
    int right = n - 1;
    int found = 0;

    while (left < right)
    {
        long long sum = (long long)arr[left] + arr[right];

        if (sum == target)
        {
            printf("Pair found: %d %d\n",
                   arr[left], arr[right]);
            found = 1;
            break;
        }
        else if (sum < target)
        {
            left++;
        }
        else
        {
            right--;
        }
    }

    if (!found)
        printf("No pair found\n");

    return 0;
}

Complexity: O(n) time and O(1) auxiliary space.

Why two pointers? Because the array is sorted. If the sum is too small, increasing the left element is the useful move. If the sum is too large, decreasing the right element is the useful move.

Problem 2: Find the maximum subarray sum

Difficulty: Medium

Concept: Kadane's algorithm.

Problem statement: Find the maximum sum of a contiguous, non-empty subarray. The array may contain negative values.

Sample input 
9
-2 1 -3 4 -1 2 1 -5 4
Sample output 
Maximum subarray sum = 6

The subarray is [4, -1, 2, 1].

Show C11 solution 
#include <stdio.h>

int main(void)
{
    int n;
    int arr[1000];

    if (scanf("%d", &n) != 1 || n <= 0 || n > 1000)
        return 1;

    for (int i = 0; i < n; i++)
    {
        if (scanf("%d", &arr[i]) != 1)
            return 1;
    }

    long long current_sum = arr[0];
    long long maximum_sum = arr[0];

    for (int i = 1; i < n; i++)
    {
        long long extend = current_sum + arr[i];

        if ((long long)arr[i] > extend)
            current_sum = arr[i];
        else
            current_sum = extend;

        if (current_sum > maximum_sum)
            maximum_sum = current_sum;
    }

    printf("Maximum subarray sum = %lld\n", maximum_sum);

    return 0;
}

Complexity: O(n) time and O(1) auxiliary space.

Interview point: Initializing both sums from the first array element is important. Initializing the maximum to zero would produce an incorrect answer when every element is negative.

Range consideration: The example uses long long for accumulation. For arbitrary input sizes and values, a production solution should also check whether the sum can exceed the range of long long.

Problem 3: Search an element using binary search

Difficulty: Easy to Medium

Concepts: Binary search, loop invariants, sorted data.

Sample input 
7
2 5 8 12 16 23 38
16
Sample output 
Element found at index 4
Show C11 solution 
#include <stdio.h>

int main(void)
{
    int n;
    int arr[1000];
    int target;

    if (scanf("%d", &n) != 1 || n < 0 || n > 1000)
        return 1;

    for (int i = 0; i < n; i++)
    {
        if (scanf("%d", &arr[i]) != 1)
            return 1;

        if (i > 0 && arr[i] < arr[i - 1])
        {
            printf("Input must be sorted\n");
            return 1;
        }
    }

    if (scanf("%d", &target) != 1)
        return 1;

    int left = 0;
    int right = n - 1;
    int found = -1;

    while (left <= right)
    {
        int mid = left + (right - left) / 2;

        if (arr[mid] == target)
        {
            found = mid;
            break;
        }
        else if (arr[mid] < target)
        {
            left = mid + 1;
        }
        else
        {
            right = mid - 1;
        }
    }

    if (found != -1)
        printf("Element found at index %d\n", found);
    else
        printf("Element not found\n");

    return 0;
}

Complexity: O(log n) time and O(1) auxiliary space.

The midpoint expression avoids the potential overflow associated with (left + right) / 2.

3. Strings and Algorithms

Problem 4: Find the longest substring without repeating characters

Difficulty: Medium to Advanced

Concept: Sliding window and character-index tracking.

Problem statement: Find the length of the longest contiguous substring containing no repeated bytes. This implementation assumes a null-terminated input consisting of bytes.

Sample input 
abcabcbb
Sample output 
Longest substring length = 3

One longest substring is "abc".

Show C11 solution 
#include <stdio.h>
#include <string.h>

int main(void)
{
    char str[1000];

    if (fgets(str, sizeof(str), stdin) == NULL)
        return 1;

    str[strcspn(str, "\n")] = '\0';

    size_t last_seen[256];

    for (size_t i = 0; i < 256; i++)
        last_seen[i] = 0;

    size_t left = 0;
    size_t best_length = 0;

    for (size_t right = 0; str[right] != '\0'; right++)
    {
        unsigned char ch = (unsigned char)str[right];

        if (last_seen[ch] > left)
            left = last_seen[ch];

        size_t current_length = right - left + 1;

        if (current_length > best_length)
            best_length = current_length;

        last_seen[ch] = right + 1;
    }

    printf("Longest substring length = %zu\n", best_length);

    return 0;
}

Complexity: O(n) time and O(1) auxiliary space, using a fixed 256-entry table.

Important detail: The table stores index + 1. Zero means the byte has not been seen. This also makes the update for the left boundary straightforward.

This is byte-oriented processing, not Unicode code-point processing.

Problem 5: Check balanced parentheses using a stack

Difficulty: Medium

Problem statement: Determine whether an expression has correctly balanced parentheses, square brackets and curly braces. Other characters are ignored.

Sample input 
{a + [b * (c + d)]}
Sample output 
Balanced
Show C11 solution 
#include <stdio.h>

#define MAX 1000

int main(void)
{
    char expression[MAX];
    char stack[MAX];
    int top = -1;

    if (fgets(expression, sizeof(expression), stdin) == NULL)
        return 1;

    int balanced = 1;

    for (int i = 0; expression[i] != '\0'; i++)
    {
        char ch = expression[i];

        if (ch == '(' || ch == '[' || ch == '{')
        {
            if (top == MAX - 1)
            {
                balanced = 0;
                break;
            }

            stack[++top] = ch;
        }
        else if (ch == ')' || ch == ']' || ch == '}')
        {
            if (top == -1)
            {
                balanced = 0;
                break;
            }

            char opening = stack[top--];

            if ((ch == ')' && opening != '(') ||
                (ch == ']' && opening != '[') ||
                (ch == '}' && opening != '{'))
            {
                balanced = 0;
                break;
            }
        }
    }

    if (top != -1)
        balanced = 0;

    if (balanced)
        printf("Balanced\n");
    else
        printf("Not balanced\n");

    return 0;
}

Complexity: O(n) time and O(n) worst-case auxiliary space.

Limitation: This checks bracket balance only. It does not parse C syntax or account for brackets inside string literals and comments.

4. Linked List Problems

Problem 6: Detect a cycle in a linked list

Difficulty: Medium

Concept: Floyd's cycle detection algorithm.

Problem statement: Determine whether a singly linked list contains a cycle. Use slow and fast pointers.

Show C11 solution 
#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

int has_cycle(const struct Node *head)
{
    const struct Node *slow = head;
    const struct Node *fast = head;

    while (fast != NULL && fast->next != NULL)
    {
        slow = slow->next;
        fast = fast->next->next;

        if (slow == fast)
            return 1;
    }

    return 0;
}

int main(void)
{
    struct Node *a = malloc(sizeof(*a));
    struct Node *b = malloc(sizeof(*b));
    struct Node *c = malloc(sizeof(*c));

    if (a == NULL || b == NULL || c == NULL)
    {
        free(a);
        free(b);
        free(c);
        return 1;
    }

    a->data = 10;
    b->data = 20;
    c->data = 30;

    a->next = b;
    b->next = c;
    c->next = b;

    printf("%s\n", has_cycle(a) ? "Cycle detected" : "No cycle");

    /* Break the cycle before freeing the allocated nodes. */
    c->next = NULL;

    free(a);
    free(b);
    free(c);

    return 0;
}
Expected output
Cycle detected

Complexity: O(n) time and O(1) auxiliary space.

Why does it work? If a cycle exists, the fast pointer eventually catches the slow pointer. If no cycle exists, the fast pointer reaches NULL.

Problem 7: Find the middle node of a linked list

Difficulty: Medium

Concept: Slow and fast pointers.

For an even number of nodes, this implementation returns the second middle node.

Example 
10 -> 20 -> 30 -> 40 -> 50
Expected result 
Middle node = 30
Show C11 solution 
#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

struct Node *find_middle(struct Node *head)
{
    struct Node *slow = head;
    struct Node *fast = head;

    while (fast != NULL && fast->next != NULL)
    {
        slow = slow->next;
        fast = fast->next->next;
    }

    return slow;
}

int main(void)
{
    int values[] = {10, 20, 30, 40, 50};
    struct Node *head = NULL;
    struct Node *tail = NULL;

    for (size_t i = 0; i < sizeof(values) / sizeof(values[0]); i++)
    {
        struct Node *node = malloc(sizeof(*node));

        if (node == NULL)
        {
            while (head != NULL)
            {
                struct Node *next = head->next;
                free(head);
                head = next;
            }
            return 1;
        }

        node->data = values[i];
        node->next = NULL;

        if (head == NULL)
            head = node;
        else
            tail->next = node;

        tail = node;
    }

    struct Node *middle = find_middle(head);

    if (middle != NULL)
        printf("Middle node = %d\n", middle->data);
    else
        printf("List is empty\n");

    while (head != NULL)
    {
        struct Node *next = head->next;
        free(head);
        head = next;
    }

    return 0;
}

Complexity: O(n) time and O(1) auxiliary space.

Interview follow-up: Explain how to return the first middle node instead when the list contains an even number of nodes.

5. Dynamic Programming

Problem 8: Find the number of ways to climb stairs

Difficulty: Medium

Concepts: Dynamic programming, recurrence relation, iterative optimization.

Problem statement: A person can climb either one or two steps at a time. Find the number of distinct ways to reach step n.

Recurrence relation:

ways(n) = ways(n - 1) + ways(n - 2)

Base cases: ways(0) = 1 and ways(1) = 1.

Sample input 
5
Sample output 
Ways = 8
Show C11 solution 
#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>

int main(void)
{
    unsigned int n;

    if (scanf("%u", &n) != 1 || n > 92U)
    {
        printf("Input out of supported range\n");
        return 1;
    }

    uint64_t previous = 1;
    uint64_t current = 1;

    if (n == 0U)
    {
        printf("Ways = 1\n");
        return 0;
    }

    for (unsigned int i = 2U; i <= n; i++)
    {
        uint64_t next = previous + current;

        previous = current;
        current = next;
    }

    printf("Ways = %" PRIu64 "\n", current);

    return 0;
}

Complexity: O(n) time and O(1) auxiliary space.

Range: The input is limited to 92 so that the result fits in a 64-bit unsigned integer.

Interview follow-up: Compare this iterative solution with a recursive solution without memoization. Explain why the naive recursive implementation repeats work.

6. Sorting and Structures

Problem 9: Sort an array of structures by marks

Difficulty: Medium

Concepts: Structures, qsort(), comparator functions.

Problem statement: Store student IDs and marks in an array of structures. Sort students by marks in descending order. If marks are equal, sort by ID in ascending order.

Show C11 solution 
#include <stdio.h>
#include <stdlib.h>

struct Student
{
    int id;
    int marks;
};

int compare_students(const void *a, const void *b)
{
    const struct Student *s1 = a;
    const struct Student *s2 = b;

    if (s1->marks > s2->marks)
        return -1;

    if (s1->marks < s2->marks)
        return 1;

    if (s1->id < s2->id)
        return -1;

    if (s1->id > s2->id)
        return 1;

    return 0;
}

int main(void)
{
    struct Student students[] =
    {
        {103, 85},
        {101, 92},
        {104, 85},
        {102, 78}
    };

    size_t count = sizeof(students) / sizeof(students[0]);

    qsort(students, count, sizeof(students[0]), compare_students);

    printf("ID       Marks\n");

    for (size_t i = 0; i < count; i++)
    {
        printf("%-8d %d\n",
               students[i].id,
               students[i].marks);
    }

    return 0;
}
Expected output 
ID       Marks
101      92
103      85
104      85
102      78

Complexity: O(n log n) average comparison count for typical qsort implementations; O(n) storage for the structure array. The C standard does not prescribe the exact sorting algorithm or its complexity.

Important: Do not return s1->marks - s2->marks from the comparator. Subtraction can overflow for extreme integer values.

Problem 10: Dynamically allocate and calculate the average of an array

Difficulty: Medium

Concepts: malloc(), free(), size_t, allocation checks and arithmetic.

Problem statement: Read n integers, allocate memory dynamically, calculate their average and release the allocated memory.

Sample input 
5
10 20 30 40 50
Sample output 
Average = 30.00
Show C11 solution 
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <stddef.h>

int main(void)
{
    size_t n;

    if (scanf("%zu", &n) != 1 || n == 0 || n > 100000U)
        return 1;

    if (n > SIZE_MAX / sizeof(int))
    {
        printf("Requested allocation is too large\n");
        return 1;
    }

    int *arr = malloc(n * sizeof(*arr));

    if (arr == NULL)
    {
        printf("Memory allocation failed\n");
        return 1;
    }

    long double sum = 0.0L;

    for (size_t i = 0; i < n; i++)
    {
        if (scanf("%d", &arr[i]) != 1)
        {
            free(arr);
            return 1;
        }

        sum += arr[i];
    }

    long double average = sum / (long double)n;

    printf("Average = %.2Lf\n", average);

    free(arr);
    arr = NULL;

    return 0;
}

Complexity: O(n) time and O(n) space.

Memory management checklist:

  • Validate the requested element count.
  • Check multiplication before allocating.
  • Check whether malloc() returned NULL.
  • Release allocated memory on every error path.
  • Free the allocation after its final use.

Numerical note: long double reduces the risk of intermediate overflow compared with a narrow integer accumulator, but it is not a universal guarantee against overflow for arbitrary input sizes and implementation ranges.

11. Common Mistakes in Coding Rounds

  • Using binary search on an unsorted array.
  • Ignoring negative values in maximum-subarray problems.
  • Using incorrect loop boundaries in sliding-window algorithms.
  • Dereferencing NULL pointers.
  • Accessing memory after it has been freed.
  • Forgetting to release dynamically allocated memory.
  • Using subtraction in qsort comparators where overflow is possible.
  • Ignoring empty arrays and empty strings.
  • Failing to consider integer overflow.
  • Providing an O(n²) solution when an O(n) or O(n log n) approach is expected.

12. Compilation and Testing

Compile each solution separately using:

gcc -std=c11 -Wall -Wextra -Wpedantic -g program.c -o program

Where supported, use sanitizers for runtime testing:

gcc -std=c11 -Wall -Wextra -Wpedantic -g \
-fsanitize=address,undefined program.c -o program
Sanitizers can detect several classes of runtime errors, but they cannot prove that an algorithm is correct or that all possible inputs are handled safely.
