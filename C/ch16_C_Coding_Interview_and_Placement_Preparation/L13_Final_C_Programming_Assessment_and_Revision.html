5. Section C – Arrays, Strings and Pointers

Four coding questions, 5 marks each.

Q21. Remove duplicate elements from a sorted array

Write a function that removes duplicates from a sorted integer array in place and returns the new logical length.

Example:

Input:  1 1 2 2 3 4 4 5
Output: 1 2 3 4 5
Length: 5

Target complexity: O(n) time and O(1) auxiliary space.

Show solution 
#include <stdio.h>
#include <stddef.h>

size_t remove_duplicates(int arr[], size_t n)
{
    if (arr == NULL || n == 0)
        return 0;

    size_t write = 1;

    for (size_t read = 1; read < n; read++)
    {
        if (arr[read] != arr[write - 1])
        {
            arr[write] = arr[read];
            write++;
        }
    }

    return write;
}

int main(void)
{
    int arr[] = {1, 1, 2, 2, 3, 4, 4, 5};
    size_t n = sizeof(arr) / sizeof(arr[0]);

    size_t new_length = remove_duplicates(arr, n);

    for (size_t i = 0; i < new_length; i++)
        printf("%d ", arr[i]);

    printf("\nLength: %zu\n", new_length);

    return 0;
}

Complexity: O(n) time and O(1) auxiliary space. The input must be sorted.

Q22. Find the first non-repeating character in a string

Write a function to return the first character that occurs exactly once. Use an unsigned-character frequency table.

Example:

Input:  swiss
Output: w
Show solution 
#include <stdio.h>
#include <stddef.h>

int first_unique(const char *str, unsigned char *result)
{
    if (str == NULL || result == NULL)
        return 0;

    size_t frequency[256] = {0};

    for (size_t i = 0; str[i] != '\0'; i++)
    {
        unsigned char c = (unsigned char)str[i];
        frequency[c]++;
    }

    for (size_t i = 0; str[i] != '\0'; i++)
    {
        unsigned char c = (unsigned char)str[i];

        if (frequency[c] == 1)
        {
            *result = c;
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    const char str[] = "swiss";
    unsigned char result;

    if (first_unique(str, &result))
        printf("First unique character: %c\n", result);
    else
        printf("No unique character\n");

    return 0;
}
First unique character: w

Complexity: O(n) time and O(1) auxiliary space for the fixed 256-entry table. This examines bytes, not Unicode characters.

Q23. Implement binary search iteratively

Write a function to find a target in a sorted integer array. Return its index or -1 if not found.

Show solution 
#include <stdio.h>
#include <stddef.h>

int binary_search(const int arr[], size_t n, int target)
{
    if (arr == NULL)
        return -1;

    size_t low = 0;
    size_t high = n;

    while (low < high)
    {
        size_t mid = low + (high - low) / 2;

        if (arr[mid] == target)
            return (int)mid;

        if (arr[mid] < target)
            low = mid + 1;
        else
            high = mid;
    }

    return -1;
}

int main(void)
{
    int arr[] = {10, 20, 30, 40, 50, 60};
    size_t n = sizeof(arr) / sizeof(arr[0]);

    printf("Index = %d\n", binary_search(arr, n, 40));

    return 0;
}
Index = 3

Complexity: O(log n) time and O(1) auxiliary space. The array must be sorted. This example assumes the array length is representable as int because the return type is int.

Q24. Swap two integers using pointers

#include <stdio.h>

void swap(int *a, int *b)
{
    if (a == NULL || b == NULL)

    int temp = *a;
    *a = *b;
    *b = temp;
}

{

    swap(&x, &y);

    printf("x = %d, y = %d\n", x, y);

    return 0;
}
x = 20, y = 10

Complexity: O(1) time and O(1) auxiliary space.

6. Section D – Functions, Recursion and Data Structures

Q25. Implement factorial using recursion.

Write a recursive function that calculates the factorial of a nonnegative integer, reporting overflow instead of producing an incorrect result.

Show solution 
#include <stdio.h>
#include <limits.h>

int factorial(unsigned int n, unsigned long long *result)
{
    if (result == NULL)
        return 0;

    unsigned long long value = 1;

    for (unsigned int i = 2; i <= n; i++)
    {
        if (value > ULLONG_MAX / i)
            return 0;

        value *= i;
    }

    *result = value;
    return 1;
}

int main(void)
{
    unsigned int n;
    unsigned long long result;

    if (scanf("%u", &n) != 1)
        return 1;

    if (factorial(n, &result))
        printf("%llu\n", result);
    else
        printf("Overflow\n");

    return 0;
}
This implementation uses an iterative calculation to avoid recursion depth risks. In an interview, explain how a recursive version would use a base case and recursive calls, and how it would need the same overflow checks.

Q26. Implement a singly linked list insertion at the beginning.

Create a node dynamically, insert it at the head, print the list and release all allocated nodes.

Show solution 
#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

int push_front(struct Node **head, int value)
{
    if (head == NULL)
        return 0;

    struct Node *node = malloc(sizeof(*node));

    if (node == NULL)
        return 0;

    node->data = value;
    node->next = *head;
    *head = node;

    return 1;
}

void display(const struct Node *head)
{
    while (head != NULL)
    {
        printf("%d -> ", head->data);
        head = head->next;
    }

    printf("NULL\n");
}

void free_list(struct Node *head)
{
    while (head != NULL)
    {
        struct Node *next = head->next;
        free(head);
        head = next;
    }
}

int main(void)
{
    struct Node *head = NULL;

    if (!push_front(&head, 30) ||
        !push_front(&head, 20) ||
        !push_front(&head, 10))
    {
        free_list(head);
        return 1;
    }

    display(head);
    free_list(head);

    return 0;
}
10 -> 20 -> 30 -> NULL

Complexity: O(1) insertion at the beginning; O(n) traversal and cleanup.

Q27. Explain stack implementation using an array.

Describe push, pop and peek operations. Discuss stack overflow and underflow conditions.

Show solution 
#include <stdio.h>
#include <stddef.h>

#define CAPACITY 5

struct Stack
{
    int data[CAPACITY];
    size_t top;
};

void initialize(struct Stack *s)
{
    if (s != NULL)
        s->top = 0;
}

int push(struct Stack *s, int value)
{
    if (s == NULL || s->top == CAPACITY)
        return 0;

    s->data[s->top++] = value;
    return 1;
}

int pop(struct Stack *s, int *value)
{
    if (s == NULL || value == NULL || s->top == 0)
        return 0;

    *value = s->data[--s->top];
    return 1;
}

int main(void)
{
    struct Stack s;
    int value;

    initialize(&s);

    push(&s, 10);
    push(&s, 20);
    push(&s, 30);

    if (pop(&s, &value))
        printf("Popped: %d\n", value);

    return 0;
}
Popped: 30

7. Section E – Dynamic Memory, Bit Manipulation and File Handling

Q28. Dynamically allocate an integer array and calculate its average.

Read an element count and the corresponding values. Allocate storage safely, validate input, calculate the average and release memory.

Show solution 
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <stddef.h>

int main(void)
{
    size_t n;

    if (scanf("%zu", &n) != 1 || n == 0)
        return 1;

    if (n > SIZE_MAX / sizeof(int))
        return 1;

    int *arr = malloc(n * sizeof(*arr));

    if (arr == NULL)
        return 1;

    long double sum = 0.0L;

    for (size_t i = 0; i < n; i++)
    {
        if (scanf("%d", &arr[i]) != 1)
        {
            free(arr);
            return 1;
        }

        sum += arr[i];
    }

    printf("Average = %.2Lf\n", sum / (long double)n);

    free(arr);
    return 0;
}
Input:
5
10 20 30 40 50

Output:
Average = 30.00

Complexity: O(n) time and O(n) allocated storage.

Q29. Set, clear and toggle a bit.

Given an unsigned integer and a bit position, demonstrate setting, clearing and toggling that bit.

Show solution 
#include <stdio.h>
#include <limits.h>

int main(void)
{
    unsigned int value = 10U;
    unsigned int position = 1U;

    if (position >= sizeof(value) * CHAR_BIT)
        return 1;

    unsigned int mask = 1U << position;

    unsigned int set_value = value | mask;
    unsigned int clear_value = value & ~mask;
    unsigned int toggle_value = value ^ mask;

    printf("Original = %u\n", value);
    printf("Set      = %u\n", set_value);
    printf("Clear    = %u\n", clear_value);
    printf("Toggle   = %u\n", toggle_value);

    return 0;
}
Original = 10
Set      = 10
Clear    = 8
Toggle   = 8

The second bit of binary 1010 is already set. Therefore, setting it changes nothing; clearing and toggling it produce 1000.

All shifts use an unsigned operand and a validated shift count.

Q30. Write a program to copy a file using binary file operations.

Use fread() and fwrite() and check file errors.

Show solution 
#include <stdio.h>

int main(void)
{
    FILE *source = fopen("input.bin", "rb");

    if (source == NULL)
    {
        perror("Cannot open source");
        return 1;
    }

    FILE *destination = fopen("output.bin", "wb");

    if (destination == NULL)
    {
        perror("Cannot open destination");
        fclose(source);
        return 1;
    }

    unsigned char buffer[4096];
    int success = 1;

    for (;;)
    {
        size_t count = fread(buffer, 1, sizeof(buffer), source);

        if (count > 0)
        {
            if (fwrite(buffer, 1, count, destination) != count)
            {
                success = 0;
                break;
            }
        }

        if (count < sizeof(buffer))
        {
            if (ferror(source))
                success = 0;

            break;
        }
    }

    if (fclose(source) != 0)
        success = 0;

    if (fclose(destination) != 0)
        success = 0;

    if (!success)
    {
        fprintf(stderr, "File copy failed\n");
        return 1;
    }

    printf("File copied successfully\n");
    return 0;
}

Run the program in a directory containing input.bin. It creates or overwrites output.bin.

8. Section F – Advanced C and Embedded C

Q31. Explain the difference between const and volatile.

Suggested answer

const restricts modification through a const-qualified access path. volatile tells the implementation that accesses to the object are observable side effects and must respect volatile access rules.

Neither qualifier independently guarantees thread safety. volatile is not a replacement for atomic operations or synchronization.

Q32. Explain the C program memory layout.

Suggested answer

A typical process may use regions commonly described as:

  • Text/code region
  • Read-only data
  • Initialized global and static data
  • BSS or zero-initialized data
  • Heap
  • Stack

This is a common implementation model, not a memory layout mandated by the C language standard.

Q33. What is structure padding?

Suggested answer

Compilers may insert padding between structure members or at the end of a structure to satisfy alignment requirements. Therefore, sizeof(structure) can exceed the sum of the member sizes.

Q34. Explain undefined behavior with examples.

Suggested answer

Undefined behavior means the C standard imposes no requirements on the result. Examples include signed integer overflow, out-of-bounds array access, invalid pointer dereferencing and division by zero.

Q35. What is the difference between a process and a thread?

Suggested answer

A process is an executing program with its own process resources and address space under a typical operating system model. Threads are execution paths within a process and typically share process resources while maintaining their own execution state.

Q36. What is the difference between polling and interrupt-driven programming?

Suggested answer

Polling repeatedly checks device status in software. Interrupt-driven programming responds to events signaled by hardware or an interrupt controller. Interrupts can reduce unnecessary checking, but require correct interrupt configuration and shared-state management.

Q37. Why should an ISR execute quickly?

Suggested answer

A lengthy ISR can increase interrupt latency, delay other time-critical events and reduce system responsiveness. A common design is to acknowledge the event, capture essential data and defer substantial processing to a task or main loop.

Q39. What is a watchdog timer?

Suggested answer

A watchdog timer detects failure to make expected progress. If software does not service the watchdog within the configured time, the hardware may reset the system or trigger another recovery action.

Q40. What is the difference between microprocessor and microcontroller?

Suggested answer

A microprocessor primarily provides the processing core and commonly uses external memory and peripherals. A microcontroller integrates a CPU, memory and peripherals on a single chip for dedicated control applications.

11. Recommended Compilation and Debugging Practice

For supported GCC or Clang environments, try runtime sanitizers:

gcc -std=c11 -Wall -Wextra -Wpedantic \
    -fsanitize=address,undefined -g program.c -o program

Run the resulting executable and test normal, boundary and invalid inputs. Sanitizers are compiler-toolchain features, not C11 language features.
