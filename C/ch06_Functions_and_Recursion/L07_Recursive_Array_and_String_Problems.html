2. Recursive Array Traversal

Array traversal means accessing array elements one by one. In recursive traversal, a function processes the current element and calls itself for the next index.

Program

#include <stdio.h>

void displayArray(const int arr[], int index, int size)
{
    if (index == size)
        return;

    printf("%d ", arr[index]);

    displayArray(arr, index + 1, size);
}

int main(void)
{
    int arr[] = {10, 20, 30, 40, 50};
    int size = (int)(sizeof(arr) / sizeof(arr[0]));

    displayArray(arr, 0, size);

    return 0;
}
Output: 
10 20 30 40 50

Execution Trace

displayArray(arr, 0, 5)
    Print arr[0] = 10
    displayArray(arr, 1, 5)
        Print arr[1] = 20
        displayArray(arr, 2, 5)
            Print arr[2] = 30
            displayArray(arr, 3, 5)
                Print arr[3] = 40
                displayArray(arr, 4, 5)
                    Print arr[4] = 50
                    displayArray(arr, 5, 5)
                        Base case: return

Time complexity: O(n)

Auxiliary call-stack space: O(n)

3. Calculate Array Sum Using Recursion

The sum of an array can be calculated by adding the current element to the sum of the remaining elements.

sum(index) = arr[index] + sum(index + 1)

Program

#include <stdio.h>

int arraySum(const int arr[], int index, int size)
{
    if (index == size)
        return 0;

    return arr[index] + arraySum(arr, index + 1, size);
}

int main(void)
{
    int arr[] = {10, 20, 30, 40, 50};
    int size = (int)(sizeof(arr) / sizeof(arr[0]));

    printf("Array sum = %d\n",
           arraySum(arr, 0, size));

    return 0;
}
Output: 
Array sum = 150

Time complexity: O(n)

Auxiliary call-stack space: O(n)

This example assumes the sum fits in int. Signed integer overflow is undefined behavior in C.

4. Find Maximum Element in an Array

The recursive function compares the current element with the maximum value found in the remaining array.

Program

#include <stdio.h>

int findMaximum(const int arr[], int index, int size)
{
    int maxRemaining;

    if (index == size - 1)
        return arr[index];

    maxRemaining = findMaximum(arr, index + 1, size);

    return arr[index] > maxRemaining
           ? arr[index] : maxRemaining;
}

int main(void)
{
    int arr[] = {12, 45, 7, 89, 23};
    int size = (int)(sizeof(arr) / sizeof(arr[0]));

    if (size > 0)
        printf("Maximum = %d\n",
               findMaximum(arr, 0, size));

    return 0;
}
Output: 
Maximum = 89

Time complexity: O(n)

Auxiliary call-stack space: O(n)

The function requires a non-empty array. The main function checks that the array contains at least one element before calling it.

6. Recursive Linear Search

Linear search checks array elements sequentially until the target is found or the array ends.

Program

#include <stdio.h>

int linearSearch(const int arr[], int index,
                 int size, int key)
{
    if (index == size)
        return -1;

    if (arr[index] == key)
        return index;

    return linearSearch(arr, index + 1, size, key);
}

int main(void)
{
    int arr[] = {10, 25, 30, 45, 50};
    int size = (int)(sizeof(arr) / sizeof(arr[0]));
    int key = 45;

    int result = linearSearch(arr, 0, size, key);

    if (result != -1)
        printf("Element found at index %d\n", result);
    else
        printf("Element not found\n");

    return 0;
}
Output: 
Element found at index 3

Best-case time complexity: O(1)

Worst-case time complexity: O(n)

Auxiliary call-stack space: O(n) worst case

7. Recursive Binary Search

Binary search repeatedly divides a sorted array into two halves and searches only the relevant half.

Binary search requires the array to be sorted. Applying it to an unsorted array does not guarantee correct results.

Program

#include <stdio.h>

int binarySearch(const int arr[], int low,
                 int high, int key)
{
    int mid;

    if (low > high)
        return -1;

    mid = low + (high - low) / 2;

    if (arr[mid] == key)
        return mid;

    if (key < arr[mid])
        return binarySearch(arr, low, mid - 1, key);

    return binarySearch(arr, mid + 1, high, key);
}

int main(void)
{
    int arr[] = {10, 20, 30, 40, 50, 60, 70};
    int size = (int)(sizeof(arr) / sizeof(arr[0]));
    int key = 50;

    int result = binarySearch(arr, 0, size - 1, key);

    if (result != -1)
        printf("Element found at index %d\n", result);
    else
        printf("Element not found\n");

    return 0;
}
Output: 
Element found at index 4

Execution Trace

Array: 10 20 30 40 50 60 70
Key: 50

low = 0, high = 6
mid = 3, arr[3] = 40
50 > 40, search right half

low = 4, high = 6
mid = 5, arr[5] = 60
50 < 60, search left half

low = 4, high = 4
mid = 4, arr[4] = 50
Element found

Time complexity: O(log n)

Auxiliary call-stack space: O(log n)

This implementation uses int indices and assumes the array size fits in int. The mid calculation avoids the overflow risk of (low + high) / 2.

8. Recursive String Traversal

A C string is a character array terminated by the null character '\0'.

Recursive traversal processes one character at a time until the null terminator is reached.

Program

#include <stdio.h>

void displayString(const char str[], int index)
{
    if (str[index] == '\0')
        return;

    putchar(str[index]);

    displayString(str, index + 1);
}

int main(void)
{
    char text[] = "HELLO";

    displayString(text, 0);
    putchar('\n');

    return 0;
}
Output: 
HELLO

Time complexity: O(n)

Auxiliary call-stack space: O(n)

9. Find String Length Using Recursion

The length of a string is the number of characters before its null terminator.

Program

#include <stdio.h>
#include <stddef.h>

size_t stringLength(const char str[])
{
    if (str[0] == '\0')
        return 0;

    return 1 + stringLength(str + 1);
}

int main(void)
{
    char text[] = "Embedded C";

    printf("Length = %zu\n", stringLength(text));

    return 0;
}
Output: 
Length = 10

Time complexity: O(n)

Auxiliary call-stack space: O(n)

10. Reverse a String Using Recursion

A string can be reversed by swapping characters at the beginning and end, then moving toward the center.

Program

#include <stdio.h>
#include <string.h>

void reverseString(char str[], size_t left, size_t right)
{
    char temp;

    if (left >= right)
        return;

    temp = str[left];
    str[left] = str[right];
    str[right] = temp;

    reverseString(str, left + 1, right - 1);
}

int main(void)
{
    char text[] = "HELLO";
    size_t length = strlen(text);

    if (length > 0)
        reverseString(text, 0, length - 1);

    printf("Reversed string = %s\n", text);

    return 0;
}
Output: 
Reversed string = OLLEH

Time complexity: O(n)

Auxiliary call-stack space: O(n)

This program modifies the original character array. The input must be a writable, null-terminated string.

11. Check String Palindrome Using Recursion

A palindrome string reads the same in both directions.

Examples: madam, level and radar.

Program

#include <stdio.h>
#include <string.h>

int isPalindrome(const char str[],
                 size_t left, size_t right)
{
    if (left >= right)
        return 1;

    if (str[left] != str[right])
        return 0;

    return isPalindrome(str, left + 1, right - 1);
}

int main(void)
{
    char text[] = "madam";
    size_t length = strlen(text);

    if (isPalindrome(text, 0, length))
        printf("Palindrome string\n");
    else
        printf("Not a palindrome string\n");

    return 0;
}
Output: 
Palindrome string

Time complexity: O(n)

Auxiliary call-stack space: O(n)

This example performs a case-sensitive comparison and includes spaces and punctuation as ordinary characters.

12. Compare Two Strings Recursively

Two strings are equal if their corresponding characters match and both terminate at the same position.

Program

#include <stdio.h>

int compareStrings(const char first[],
                   const char second[])
{
    if (first[0] != second[0])
        return 0;

    if (first[0] == '\0')
        return 1;

    return compareStrings(first + 1, second + 1);
}

int main(void)
{
    char first[] = "Embedded";
    char second[] = "Embedded";

    if (compareStrings(first, second))
        printf("Strings are equal\n");
    else
        printf("Strings are different\n");

    return 0;
}
Output: 
Strings are equal

Time complexity: O(n)

Auxiliary call-stack space: O(n)

14. Common Mistakes

  1. Using an incorrect array index in the recursive call.
  2. Forgetting to stop when the array boundary is reached.
  3. Calling binary search on an unsorted array.
  4. Using an incorrect midpoint or range update.
  5. Forgetting that a C string ends with '\0'.
  6. Accessing characters outside the valid string bounds.
  7. Trying to modify a string stored in a read-only string literal.
  8. Ignoring excessive recursion depth for large arrays or strings.

16. Technical Interview Questions

Q1. How can an array be traversed recursively?

Process the current index and call the function again with the next index until the array boundary is reached.

Q2. What is the base case for array traversal?

A common base case is index == size.

Q3. What is the difference between recursive linear and binary search?

Linear search checks elements sequentially. Binary search repeatedly halves a sorted search range.

Q4. What is the time complexity of recursive binary search?

O(log n).

Q5. Why must binary search use a sorted array?

Sorting allows the algorithm to determine which half can contain the target.

Q6. How does a recursive function detect the end of a C string?

It checks for the null terminator '\0'.

Q7. How can a string be reversed recursively?

Swap the first and last characters, then recursively process the smaller inner range.

Q8. What is the base case for recursive palindrome checking?

When the left index is greater than or equal to the right index, all required character pairs have matched.

Q9. What is the auxiliary space complexity of recursive array traversal?

O(n) due to the recursive call stack.

Q10. Why can recursion be problematic for large arrays?

Deep recursive calls can require substantial stack space and may exhaust available stack resources.
